12. AOI 兴趣区域系统 · 九宫格 + 空间哈希

12.1 设计动机

纯九宫格（Grid）在实体分布极不均匀时退化：主城广场数千人挤在少数格子中，一次查询退化为全量遍历； 野外大部分格子空置，浪费内存。本方案采用 九宫格 + 空间哈希混合： 九宫格做第一层索引。当格子内实体 ≥ 200 时自动升级为空间哈希（16 子格精细查询），实体 ≤ 50 自动降级。

12.2 坐标系统与格子参数

AOI 基于 水平面（x, z） 做空间索引。y 轴（高度）不参与 AOI 格子划分—— 高度校验通过单独的视野范围（|y1 - y2| < viewRangeY）做过滤。 这和大多数 MMO 的做法一致（WoW、TrinityCore 都用 2D 水平面 AOI）。

参数默认值说明
kGridSize50.0m格子边长。应 ≈ viewRadius，使九宫格正好覆盖视野。
kViewRadiusXZ100.0m水平视野半径。AOI 查询范围 = 实体所在格 ± 2 圈。
kViewRadiusY50.0m高度视野阈值。AOI 结果出来后按 |dy| 过滤。
kHotThreshold200格子内实体 ≥ 200 → 升级为热点模式。
kCoolThreshold50格子内实体 ≤ 50 → 降级回普通模式。
kSubGrids4每维 4 份，热点模式内部共 16 个子格。
kMaxPacketSize64KB与 S7 一致，单次 AOI 广播包上限。

12.3 核心数据结构

// GridAoiSystem.h

struct CellIndex { int x, z; };        // 格子坐标（整数索引）
struct SubCellKey { int sx, sz; };     // 子格坐标（0..kSubGrids-1）

// 热-点模式：子格 → 实体列表
using SubCellMap = std::unordered_map<uint64_t, std::vector<Entity>>;

class GridCell {
public:
    std::vector<Entity> entities;         // 普通模式：直接存实体列表
    SubCellMap hotSubCells;          // 热点模式：子格 → 实体列表
    bool isHot = false;

    static constexpr size_t kHotThreshold = 200;
    static constexpr size_t kCoolThreshold = 50;
    static constexpr uint32_t kSubGrids = 4;  // 每维 4 份，共 16 子格
};

class GridAoiSystem {
public:
    GridAoiSystem(uint32_t sceneId, float gridSize, float viewRadiusXZ, float viewRadiusY);

    // —— 实体位置更新（每 Tick 调用）——
    void UpdateEntityPosition(Entity entity, float x, float y, float z);

    // —— 查询 ——
    std::vector<Entity> GetWatchers(Entity entity);  // 返回 entity 的 AOI 范围内所有其他实体（去重+排除自身）
    std::vector<Entity> GetEntitiesInRadius(float x, float z, float radius);

    // —— 事件 ——
    std::vector<Entity> GetEnteredEntities(Entity entity);   // 本 Tick 新进入 entity 视野的实体
    std::vector<Entity> GetLeftEntities(Entity entity);      // 本 Tick 离开 entity 视野的实体

    // —— 热点管理 ——
    void CheckHotSpotDowngrade();  // 每 Tick 检查是否有热点格子可以降级

private:
    // 位置 → 格子索引
    CellIndex GetCellIndex(float x, float z) const;
    bool     IsValidCell(const CellIndex& c) const;

    // 子格索引
    SubCellKey GetSubCellKey(float x, float z, const CellIndex& cell) const;
    uint64_t   EncodeSubCellKey(const SubCellKey& key, const CellIndex& cell) const;

    // 升降级
    void UpgradeToHotCell(const CellIndex& cell);
    void DowngradeFromHotCell(const CellIndex& cell);

    // 添加/移除实体
    void AddToCell(Entity entity, const CellIndex& cell, float x, float z);
    void RemoveFromCell(Entity entity, const CellIndex& cell);

    uint32_t _sceneId;
    float    _gridSize;
    float    _subGridSize;    // _gridSize / kSubGrids
    float    _viewRadiusXZ;
    float    _viewRadiusY;
    int      _viewRangeCells;  // ceil(_viewRadiusXZ / _gridSize)

    // 2D 网格（稀疏——只分配有实体的格子）
    std::unordered_map<uint64_t, GridCell> _cells;  // key = EncodeCellKey(x,z)
    std::unordered_map<Entity, CellIndex> _entityCells;   // entity → 当前格子

    // 各实体在上个 Tick 的可见集——用于计算 enter/leave 事件
    std::unordered_map<Entity, std::unordered_set<Entity>> _prevVisible;
};

12.4 核心算法

12.4.1 实体位置更新

void GridAoiSystem::UpdateEntityPosition(Entity entity, float x, float y, float z) {
    auto newCell = GetCellIndex(x, z);  // 水平面: cellX = floor(x / _gridSize), cellZ = floor(z / _gridSize)
    auto it = _entityCells.find(entity);

    if (it == _entityCells.end()) {
        // 新实体——首次添加到 AOI
        AddToCell(entity, newCell, x, z);
        _entityCells[entity] = newCell;
    }
    else if (it->second.x != newCell.x || it->second.z != newCell.z) {
        // 跨格移动
        auto oldCell = it->second;
        RemoveFromCell(entity, oldCell);
        AddToCell(entity, newCell, x, z);
        _entityCells[entity] = newCell;
    }
    else if (_cells[EncodeCellKey(newCell)].isHot) {
        // 同一热点格子内移动——更新子格归属
        UpdateHotCellPosition(entity, x, z, newCell);
    }

    _entityPositions[entity] = {x, y, z};
}

// 添加实体到格子，自动检测热点阈值
void GridAoiSystem::AddToCell(Entity entity, const CellIndex& cell, float x, float z) {
    auto key = EncodeCellKey(cell);  // key = (uint64_t(cellX) << 32) | uint32_t(cellZ)
    auto& c = _cells[key];  // 格子不存在时自动创建

    if (c.isHot) {
        auto subKey = GetSubCellKey(x, z, cell);
        c.hotSubCells[EncodeSubCellKey(subKey, cell)].push_back(entity);
    } else {
        c.entities.push_back(entity);
        if (c.entities.size() >= GridCell::kHotThreshold)
            UpgradeToHotCell(cell);
    }
}

12.4.2 GetWatchers — 九宫格查询

std::vector<Entity> GridAoiSystem::GetWatchers(Entity entity) {
    auto itPos = _entityPositions.find(entity);
    if (itPos == _entityPositions.end()) return {};

    auto cellIt = _entityCells.find(entity);
    if (cellIt == _entityCells.end()) return {};

    auto& pos  = itPos->second;
    auto& cell = cellIt->second;

    std::unordered_set<Entity> resultSet;  // 自动去重

    // 遍历九宫格 (±_viewRangeCells 格)
    for (int dx = -_viewRangeCells; dx <= _viewRangeCells; dx++) {
        for (int dz = -_viewRangeCells; dz <= _viewRangeCells; dz++) {
            CellIndex neighbor{cell.x + dx, cell.z + dz};

            auto cit = _cells.find(EncodeCellKey(neighbor));
            if (cit == _cells.end()) continue;  // 空格子，跳过

            auto& gc = cit->second;
            if (gc.isHot) {
                // 热点模式：只查与 entity 所在子格相邻的子格（九宫格内的九宫格）
                auto subKey = GetSubCellKey(pos.x, pos.z, neighbor);
                for (int sdx = -1; sdx <= 1; sdx++)
                    for (int sdz = -1; sdz <= 1; sdz++) {
                        auto hit = gc.hotSubCells.find(
                            EncodeSubCellKey({subKey.sx + sdx, subKey.sz + sdz}, neighbor));
                        if (hit != gc.hotSubCells.end())
                            resultSet.insert(hit->second.begin(), hit->second.end());
                    }
            } else {
                // 普通模式：直接取全部
                resultSet.insert(gc.entities.begin(), gc.entities.end());
            }
        }
    }

    // 排除自身
    resultSet.erase(entity);

    return {resultSet.begin(), resultSet.end()};
}

12.4.3 AOI 事件计算 (Enter / Leave)

每 Tick 在各实体的 GetWatchers 结果和上一 Tick 的可见集之间做 diff，计算出新进入视野和离开视野的实体列表， 通过 DasLang 桥接 API 投递事件给脚本处理（发 EntityCreateNtf / EntityDestroyNtf 给对应客户端）。 

// C++ 引擎在 SystemAOI 中调用，每 Tick
void GridAoiSystem::FlushAoiEvents(Scene& scene) {
    for (auto& [entity, prevSet] : _prevVisible) {
        auto nowSet = GetWatchersAsSet(entity);
        auto entered = std::set_difference(nowSet, prevSet);
        auto left    = std::set_difference(prevSet, nowSet);

        for (auto e : entered)
            OnEntityEnterAoi(entity, e);  // → DasLang 回调
        for (auto e : left)
            OnEntityLeaveAoi(entity, e);

        prevSet = std::move(nowSet);
    }
}

12.4.4 热点升级 / 降级

void GridAoiSystem::UpgradeToHotCell(const CellIndex& cellIdx) {
    auto& cell = _cells[EncodeCellKey(cellIdx)];
    cell.isHot = true;

    // 将所有实体按精细坐标分配到 16 个子格
    for (auto e : cell.entities) {
        auto itPos = _entityPositions.find(e);
        if (itPos == _entityPositions.end()) continue;
        auto subKey = GetSubCellKey(itPos->second.x, itPos->second.z, cellIdx);
        cell.hotSubCells[EncodeSubCellKey(subKey, cellIdx)].push_back(e);
    }

    cell.entities.clear();  // 释放普通列表
}

void GridAoiSystem::DowngradeFromHotCell(const CellIndex& cellIdx) {
    auto& cell = _cells[EncodeCellKey(cellIdx)];

    // 收集所有子格的实体回到普通列表
    for (auto& [key, entities] : cell.hotSubCells)
        cell.entities.insert(cell.entities.end(), entities.begin(), entities.end());

    auto [it, _] = cell.hotSubCells.clear();  // 或直接 cell.hotSubCells = {} 触发析
    auto temp = SubCellMap{};
    cell.hotSubCells.swap(temp);             // 释放内存
    cell.isHot = false;
}

12.5 与 Scene / ECS 集成

每个 Scene 持有自己的 GridAoiSystem。Scene::OnEntityMove 同时更新 EnTT Position 组件和 AOI 位置。 ecs_stage "aoi" 阶段调用 Scene::FlushAoiEvents，diff 计算 Enter/Leave 事件并投递给 DasLang 脚本。 

// Scene.h — AOI 集成
class Scene {
public:
    void OnEntityMove(Entity entity, float x, float y, float z) {
        // 1. 更新 EnTT Position 组件
        auto& pos = _ecsRegistry.get<Position>(entt::entity(entity.EntityId()));
        pos.x = x; pos.y = y; pos.z = z;

        // 2. 更新 AOI 系统
        _aoi.UpdateEntityPosition(entity, x, y, z);
    }

    void FlushAoiEvents() {
        _aoi.FlushAoiEvents(*this);  // 内部调 OnEntityEnterAoi / OnEntityLeaveAoi → DasLang
    }

    GridAoiSystem& GetAoi() { return _aoi; }

private:
    entt::registry _ecsRegistry;
    GridAoiSystem _aoi;
};

12.6 DasLang 桥接 API

// C++ 侧注册给 DasLang
GetAoiWatchers(sceneId: uint32; entityId: Entity): array<Entity>
GetEntitiesInAoiRadius(sceneId: uint32; x: float; z: float; radius: float): array<Entity>

// AOI 事件回调（C++ 引擎调用）
OnEntityEnterAoi(observer: Entity; target: Entity)   // observer 的视野中进入了 target
OnEntityLeaveAoi(observer: Entity; target: Entity)   // target 离开 observer 的视野

// DasLang 侧使用——发送网络消息给客户端
def OnEntityEnterAoi(observer : Entity; target : Entity)
{
    let msg = EntityCreateNtf{ entityId = target, entityType = GetEntityType(target) }
    SendToClient(observer, MSG_ENTITY_CREATE_NTF, SerializeProto(msg))
}

def OnEntityLeaveAoi(observer : Entity; target : Entity)
{
    let msg = EntityDestroyNtf{ entityId = target }
    SendToClient(observer, MSG_ENTITY_DESTROY_NTF, SerializeProto(msg))
}

12.7 3D 兼容说明

AOI 水平面（x, z）用九宫格 + 空间哈希做空间索引。高度（y）在 GetWatchers 结果集上做后过滤： |entity.y - observer.y| < _viewRadiusY。这不是性能瓶颈——AOI 结果集通常在 50-200 个实体， 遍历一次做高度过滤是 O(n) 的，成本远低于格子索引本身。

12.8 性能特征

操作复杂度说明
UpdateEntityPositionO(1) 或 O(log n)hash map 查格子 + vector push。跨格时为旧格 O(1) 删除（无序遍历交换）。
GetWatchersO(K × M)K = 九宫格格数（通常 9 或 25），M = 每格平均实体数。热点模式为子格查询。
FlushAoiEventsO(N × M)N = 实体数。每 Tick 对所有有 AOI 关注的实体做可见集 diff。
UpgradeToHotCellO(E)E = 格子内实体数（一次性）。触发频率极低（每格子一次）。
设计验证：本方案参考了 WoW (TrinityCore) 的九宫格 AOI、MOBA 视野筛减技术（广播量可降低 90%+）、 及空间哈希的轻量化空间索引实践。与纯九宫格相比，热点模式在主城广场场景下将查询范围从 2000 实体缩小到 ~50 实体的子格范围， 避免了 O(n²) 退化。
