物品与背包系统 · Bag + Equipment + Trade

1. 设计概述

物品系统采用 扁平化物品实例 设计：每个物品是独立实例（instanced item），拥有唯一的 itemInstanceId。 这支持了强化、附魔、随机属性等个性化操作。背包是物品实例的容器，装备是背包的特化视图。 物品操作（移动/拆分/堆叠/删除）走 全权服务器校验，客户端只是提交操作请求 + 渲染。

3.1 物品实例 ID 生成

// C++ 侧——全服唯一 ID 生成器
class ItemInstanceIdGenerator {
public:
    static uint64_t Generate() {
        // 高 32 位 = 时间戳（秒），低 32 位 = 自增序号
        auto ts = uint64_t(time(nullptr)) << 32;
        return ts | _counter.fetch_add(1);
    }
private:
    static inline std::atomic<uint32_t> _counter{0};
};

5. 背包操作——全权服务器校验

5.1 移动物品

// MSG_BAG_MOVE_REQ(fromSlot, toSlot) → MSG_BAG_MOVE_RSP
def HandleBagMove(player : Entity; fromSlot : int; toSlot : int)
{
    ecs_edit(player) $(var cmp)
    {
        // 校验：槽位范围
        if (fromSlot < 0 || fromSlot >= cmp.inventory.maxSlots) {
            SendBagError(player, BAG_ERR_INVALID_SLOT); return
        }
        if (toSlot < 0 || toSlot >= cmp.inventory.maxSlots) {
            SendBagError(player, BAG_ERR_INVALID_SLOT); return
        }

        let fromItem = cmp.inventory.slots[fromSlot].item
        let toItem   = cmp.inventory.slots[toSlot].item

        if (fromItem == null) {
            SendBagError(player, BAG_ERR_EMPTY_SLOT); return
        }

        // 堆叠逻辑：同 itemId + 可堆叠 + 未超 maxStack
        if (toItem != null && toItem.itemId == fromItem.itemId) {
            let template = g_itemTemplates.Find(fromItem.itemId)
            if (template != null && template.max_stack > 1) {
                let space = template.max_stack - toItem.count
                let move = min(fromItem.count, space)
                toItem.count += move
                fromItem.count -= move
                if (fromItem.count <= 0) {
                    cmp.inventory.slots[fromSlot].item = null
                }
                MarkInventoryDirty(player)
                SendBagMoveRsp(player, BAG_OK); return
            }
        }

        // 交换位置
        cmp.inventory.slots[toSlot].item   = fromItem
        cmp.inventory.slots[fromSlot].item = toItem

        MarkInventoryDirty(player)
        SendBagMoveRsp(player, BAG_OK)
    }
}

5.2 拆分物品

def HandleBagSplit(player : Entity; fromSlot : int; count : int; toSlot : int)
{
    ecs_edit(player) $(var cmp)
    {
        let fromItem = cmp.inventory.slots[fromSlot].item
        if (fromItem == null || count <= 0 || count >= fromItem.count) {
            SendBagError(player, BAG_ERR_INVALID_COUNT); return
        }

        // 创建拆分副本并减少源数量
        let splitCopy = CloneItemInstance(fromItem)
        splitCopy.count = count
        fromItem.count -= count

        // 目标槽必须为空
        if (cmp.inventory.slots[toSlot].item != null) {
            SendBagError(player, BAG_ERR_SLOT_OCCUPIED); return
        }

        cmp.inventory.slots[toSlot].item = splitCopy
        MarkInventoryDirty(player)
        SendBagMoveRsp(player, BAG_OK)
    }
}

5.3 使用物品

def HandleBagUse(player : Entity; slot : int)
{
    ecs_edit(player) $(var cmp)
    {
        let item = cmp.inventory.slots[slot].item
        if (item == null) { SendBagError(player, BAG_ERR_EMPTY_SLOT); return }

        let template = g_itemTemplates.Find(item.itemId)
        if (template == null) { SendBagError(player, BAG_ERR_ITEM_NOT_FOUND); return }

        // 装备类型 → 穿装备
        if (template.item_type == 1) {
            EquipItem(player, slot)
            return
        }

        // 消耗品 → 使用脚本 + 减少数量
        if (template.use_script != "") {
            CallItemUseScript(template.use_script, player, item)
        }

        item.count -= 1
        if (item.count <= 0) {
            cmp.inventory.slots[slot].item = null
        }

        // 绑定检查——使用绑定
        if (template.bind_type == 3 && item.bindStatus == 0) {
            item.bindStatus = 1
            item.bindToPlayerId = GetPlayerId(player)
        }

        MarkInventoryDirty(player)
        SendBagUseRsp(player, BAG_OK)
    }
}

6. 装备系统

6.1 穿装备

def EquipItem(player : Entity; bagSlot : int)
{
    ecs_edit(player) $(var cmp)
    {
        let item = cmp.inventory.slots[bagSlot].item
        let template = g_itemTemplates.Find(item.itemId)

        // 校验
        if (template.equip_slot < 0) { SendBagError(player, BAG_ERR_NOT_EQUIPPABLE); return }
        if (GetPlayerLevel(player) < template.equip_level_min) {
            SendBagError(player, BAG_ERR_LEVEL_TOO_LOW); return
        }
        if (template.equip_class_req != 0 && template.equip_class_req != GetPlayerClass(player)) {
            SendBagError(player, BAG_ERR_CLASS_MISMATCH); return
        }

        // 交换：如果有旧装备，换到背包空位
        let oldEquip = cmp.equipmentSlots.slots[template.equip_slot]

        if (oldEquip != null) {
            // 旧装备回到背包
            let emptySlot = FindEmptyBagSlot(cmp.inventory)
            if (emptySlot < 0) { SendBagError(player, BAG_ERR_BAG_FULL); return }
            cmp.inventory.slots[emptySlot].item = oldEquip
        }

        // 穿上新装备
        cmp.equipmentSlots.slots[template.equip_slot] = item
        cmp.inventory.slots[bagSlot].item = null

        // 装备绑定
        if (template.bind_type == 2 && item.bindStatus == 0) {
            item.bindStatus = 1
            item.bindToPlayerId = GetPlayerId(player)
        }

        // 属性重算
        ecs_emplace(player, type<StatsDirtyTag>, StatsDirtyTag{})

        MarkInventoryDirty(player)
        SendEquipRsp(player, BAG_OK)
    }
}

6.2 脱装备

def UnequipItem(player : Entity; equipSlot : int)
{
    ecs_edit(player) $(var cmp)
    {
        let item = cmp.equipmentSlots.slots[equipSlot]
        if (item == null) { SendBagError(player, BAG_ERR_EMPTY_SLOT); return }

        let emptySlot = FindEmptyBagSlot(cmp.inventory)
        if (emptySlot < 0) { SendBagError(player, BAG_ERR_BAG_FULL); return }

        cmp.inventory.slots[emptySlot].item = item
        cmp.equipmentSlots.slots[equipSlot] = null

        ecs_emplace(player, type<StatsDirtyTag>, StatsDirtyTag{})
        MarkInventoryDirty(player)
        SendUnequipRsp(player, BAG_OK)
    }
}

7. 交易系统

交易系统采用 双确认锁步 模型，防止各种边界竞争条件。

7.1 交易流程

Player A                               Server                            Player B
  │ MSG_TRADE_REQ(B)                     │                                  │
  │─────────────────────────────────────▶│                                  │
  │                                      │ MSG_TRADE_REQ_NTF(A)            │
  │                                      │─────────────────────────────────▶│
  │                                      │         MSG_TRADE_RSP(accept)   │
  │                                      │◀─────────────────────────────────│
  │                                      │                                  │
  │  [双方进入 TradeState]                │                                  │
  │                                      │ 创建 TradeSession{A, B}          │
  │                                      │                                  │
  │ MSG_TRADE_ADD_ITEM(slot, count)      │                                  │
  │─────────────────────────────────────▶│                                  │
  │                                      │ 物品移动到 TradeSession.AItems   │
  │ MSG_TRADE_UPDATE_NTF                 │  → 同步给对方                   │
  │◀─────────────────────────────────────│ MSG_TRADE_UPDATE_NTF            │
  │                                      │─────────────────────────────────▶│
  │                                                                        │
  │ (双方可以多次添加/移除物品/金币)        │                                  │
  │                                                                        │
  │ MSG_TRADE_CONFIRM_REQ                │                                  │
  │─────────────────────────────────────▶│                                  │
  │                                      │ A._confirmed = true              │
  │                                      │                                  │
  │                                      │          MSG_TRADE_CONFIRM_REQ   │
  │                                      │◀─────────────────────────────────│
  │                                      │                                  │
  │                                      │ [双方都确认]                      │
  │                                      │ 1. 验货：物品仍在 A/B 的 TradeItems 中
  │                                      │ 2. 金额校验                      │
  │                                      │ 3. 空间检查（接收方背包够不够）    │
  │                                      │ 4. 原子交换                      │
  │                                      │ 5. 写 DB（立即同步）              │
  │ MSG_TRADE_COMPLETE_NTF               │                                  │
  │◀─────────────────────────────────────│ MSG_TRADE_COMPLETE_NTF           │
  │                                      │─────────────────────────────────▶│

  任一方在 Confirm 前可以 MSG_TRADE_CANCEL_REQ
  → TradeSession 销毁 → 物品返回各自背包

7.2 TradeSession 管理（C++ 侧）

// TradeSession.h — 交易会话（C++ 引擎管理，非 ScriptComponent）
class TradeSession {
public:
    struct Party {
        Entity player;
        std::vector<ItemInstance> items;    // 从背包移出的物品
        int64_t gold = 0;
        bool confirmed = false;
    };

    Party _a, _b;
    std::chrono::steady_clock::time_point _startTime;
    static constexpr auto kTimeout = std::chrono::minutes(5);

    void AddItem(Entity player, ItemInstance item);
    void RemoveItem(Entity player, uint64_t instanceId);
    void SetGold(Entity player, int64_t amount);
    void Confirm(Entity player);

    // 双确认后原子执行
    bool ExecuteTrade();  // 返回 false = 交易失败（空间不足等）
    void Cancel();

private:
    Party& GetParty(Entity player);
    bool ValidateItems(const Party& party);  // 物品未被外部修改
    bool CheckInventorySpace(Entity receiver, const std::vector<ItemInstance>& items);
};

8. 物品掉落

// 掉落生成——怪物死亡时调用
def GenerateDrops(monster : Entity; killer : Entity)
{
    let templateId = GetMonsterTemplateId(monster)
    let dropTable = g_dropTables.Find(templateId)
    if (dropTable == null) return

    let pos = GetPosition(monster)
    var drops : array<ItemInstance>

    for entry in dropTable.entries {
        if (RandomFloat(0.0, 1.0) < entry.dropChance) {
            let item = CreateItemInstance(entry.itemId, entry.minCount, entry.maxCount)
            if (item != null) {
                // 生成随机属性
                RollRandomStats(item, entry)
                drops |> push(item)
            }
        }
    }

    // 生成地面掉落物 Entity（EnTT）
    for item in drops {
        let dropEntity = ecs_create(GetSceneId(monster))
        ecs_edit(dropEntity) $(var cmp) {
            cmp.position = pos
            cmp.groundItem = GroundItem{
                item         = item,
                ownerPlayer  = killer,        // 30s 内只有 killer 可见
                dropTime     = GetTickTime(),
                expireTime   = GetTickTime() + 120.0  // 2 分钟后消失
            }
        }
    }
}

// 拾取——MSG_LOOT_REQ(entityId)
def HandleLoot(player : Entity; dropEntity : Entity)
{
    if (DistanceBetween(player, dropEntity) > 5.0) {
        SendLootRsp(player, LOOT_ERR_TOO_FAR); return
    }

    ecs_edit(player) $(var cmp) {
    ecs_edit(dropEntity) $(var dropCmp)
    {
        let ground = dropCmp.groundItem

        // 归属检查（30s 内只有 killer 能捡）
        if (ground.ownerPlayer != InvalidEntity && ground.ownerPlayer != player) {
            if (GetTickTime() - ground.dropTime < 30.0) {
                SendLootRsp(player, LOOT_ERR_NOT_OWNER); return
            }
        }

        // 空间检查
        let emptySlot = FindEmptyBagSlot(cmp.inventory)
        if (emptySlot < 0) { SendLootRsp(player, LOOT_ERR_BAG_FULL); return }

        // 拾取绑定
        let template = g_itemTemplates.Find(ground.item.itemId)
        if (template != null && template.bind_type == 1) {
            ground.item.bindStatus = 1
            ground.item.bindToPlayerId = GetPlayerId(player)
        }

        // 移入背包
        cmp.inventory.slots[emptySlot].item = ground.item
        MarkInventoryDirty(player)

        // 销毁掉落物
        ecs_destroy(dropEntity)

        SendLootRsp(player, LOOT_OK)
    }}
}

10. 背包持久化策略

操作写入策略
物品移动/拆分/使用立即写 player_inventory 表（物品变更双写保数据不丢）
穿/脱装备立即写 player_inventory + player_equipment
交易完成立即写双方背包（事务性）
掉落拾取立即写，如果在副本内则 30s 延迟写入（副本崩溃回滚时可恢复）
