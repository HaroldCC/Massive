Buff/Debuff 系统 · Modifier + Tick + Event 三部曲

1. 设计概述

Buff 系统采用 Modifier 修饰器模式。每个 Buff 实例包含一组 Modifier（修改 BattleStats 的加减乘除）， 由 BuffSystem 在每 Tick 统一结算。Modifier 在属性重算阶段统一应用， 而非每个 Buff 单独写公式——这保证了 属性计算的确定性（加法先于乘法，不受 Buff 添加顺序影响）。

2. 核心数据结构

2.1 脚本组件（BlobStorage——低频、业务逻辑重）

// Buff/BuffList defs —— DasLang 脚本组件

// 修饰器类型
[script_component]
struct StatModifier {
    stat      : StatType          // 修改的属性
    op        : ModOp             // 操作类型
    value     : float             // 数值
}

enum StatType {
    STAT_ATTACK, STAT_DEFENSE, STAT_MAXHP, STAT_MAXMP,
    STAT_CRIT_RATE, STAT_CRIT_DMG, STAT_DODGE, STAT_HIT,
    STAT_ATTACK_SPEED, STAT_MOVE_SPEED,
    STAT_DAMAGE_INCREASE, STAT_DAMAGE_REDUCTION,
    STAT_HEAL_INCREASE, STAT_HEAL_REDUCTION
}

enum ModOp {
    MOD_ADD,           // 加法：+value
    MOD_MUL,           // 乘法：×(1 + value/10000)  value=500 → ×1.05
    MOD_SET_MIN,       // 保底：max(current, value)
    MOD_SET_MAX        // 封顶：min(current, value)
}

// Buff 触发时机
enum BuffTrigger {
    TRIGGER_ON_APPLY,         // 施加时触发一次（瞬发 Buff）
    TRIGGER_PERIODIC,         // 每隔 N 秒触发一次
    TRIGGER_ON_DAMAGE_DEALT,  // 造成伤害时触发
    TRIGGER_ON_DAMAGE_TAKEN,  // 受到伤害时触发
    TRIGGER_ON_HEAL_DEALT,    // 造成治疗时触发
    TRIGGER_ON_HEAL_TAKEN,    // 受到治疗时触发
    TRIGGER_ON_KILL,          // 击杀时触发
    TRIGGER_ON_DEATH,         // 死亡时触发
    TRIGGER_ON_SKILL_CAST,    // 施放技能时触发
}

// Buff 实例——每个实体身上的 1 个 Buff
[script_component]
struct BuffInstance {
    buffId        : uint32          // Buff 模板 ID → ConfigTable 查参数
    casterEntity  : Entity          // 施法者
    startTime     : float           // 施加时间 (TickTime)
    duration      : float           // 持续时长（秒），0 = 永久
    remainingTime : float           // 剩余时间（秒）
    maxStacks     : int             // 最大层数
    currentStacks : int             // 当前层数
    modifiers     : array<StatModifier>  // 属性修饰器列表（从模板拷贝）
    period        : float           // 周期性触发间隔（秒），0 = 非周期
    timeSinceLastTick : float       // 距离上次 Tick 的时间
}

// BuffList——实体持有的所有 Buff 列表
[script_component]
struct BuffList {
    buffs         : array<BuffInstance>
    dirty         : bool                // 需要重新收集 Modifier
    cachedMods    : array<StatModifier> // 缓存：所有 Buff 的 Modifier 汇总
}

3. BuffSystem 核心逻辑（DasLang）

3.1 施加 Buff

def ApplyBuff(target : Entity; caster : Entity; buffId : uint32; stacks : int = 1) : bool
{
    let template = g_buffTemplates.Find(buffId)
    if (template == null) return false

    var applied = false
    ecs_edit(target) $(var cmp)
    {
        // 检查是否已存在同 ID Buff
        for var existing in cmp.buffList.buffs {
            if (existing.buffId == buffId) {
                // 叠加：刷新持续时间 + 增加层数（不超过 maxStacks）
                existing.duration      = template.duration_sec
                existing.remainingTime = template.duration_sec
                existing.currentStacks = min(existing.currentStacks + stacks, template.max_stacks)
                existing.startTime     = GetTickTime()
                cmp.buffList.dirty = true
                applied = true
                break
            }
        }

        if (!applied) {
            // 新 Buff——构造实例
            let mods : array<StatModifier>
            for m in template.modifiers {
                mods |> push(StatModifier{stat = m.stat, op = m.op, value = m.value})
            }

            let inst = BuffInstance{
                buffId        = buffId,
                casterEntity  = caster,
                startTime     = GetTickTime(),
                duration      = template.duration_sec,
                remainingTime = template.duration_sec,
                maxStacks     = template.max_stacks,
                currentStacks = stacks,
                modifiers     = mods,
                period        = template.period_sec,
                timeSinceLastTick = 0.0
            }
            cmp.buffList.buffs |> push(inst)
            cmp.buffList.dirty = true
        }
    }

    // 触发 on_apply 脚本
    if (template.on_apply_script != "") {
        CallBuffScript(template.on_apply_script, target, caster, buffId)
    }

    return true
}

3.2 移除 Buff

def RemoveBuff(target : Entity; buffId : uint32) : bool
{
    let template = g_buffTemplates.Find(buffId)
    var removed = false

    ecs_edit(target) $(var cmp)
    {
        for var i in range(cmp.buffList.buffs |> length()) {
            if (cmp.buffList.buffs[i].buffId == buffId) {
                let inst = cmp.buffList.buffs[i]
                cmp.buffList.buffs |> erase(i)
                cmp.buffList.dirty = true
                removed = true

                // 触发 on_remove 脚本
                if (template != null && template.on_remove_script != "") {
                    CallBuffScript(template.on_remove_script, target, inst.casterEntity, buffId)
                }
                break
            }
        }
    }
    return removed
}

def RemoveAllDebuffs(target : Entity)
{
    ecs_edit(target) $(var cmp)
    {
        for var i = cmp.buffList.buffs |> length() - 1; i >= 0; i-- {
            let template = g_buffTemplates.Find(cmp.buffList.buffs[i].buffId)
            if (template != null && template.buff_type == 1) {  // Debuff
                if (template.on_remove_script != "") {
                    CallBuffScript(template.on_remove_script, target, cmp.buffList.buffs[i].casterEntity, cmp.buffList.buffs[i].buffId)
                }
                cmp.buffList.buffs |> erase(i)
            }
        }
        cmp.buffList.dirty = true
    }
}

3.3 每 Tick 更新

// SystemBuffTick — ecs_stage "buff" 中调用
[export]
def SystemBuffTick(sceneId : uint32; dt : float)
{
    ecs_query(sceneId;
        bl : BuffList&;
        without DeadTag
    ) {
        for var i = bl.buffs |> length() - 1; i >= 0; i-- {
            let template = g_buffTemplates.Find(bl.buffs[i].buffId)
            if (template == null) {
                bl.buffs |> erase(i)
                continue
            }

            // 1. 时长递减
            if (bl.buffs[i].duration > 0.0) {
                bl.buffs[i].remainingTime -= dt
                if (bl.buffs[i].remainingTime <= 0.0) {
                    // 到期移除
                    if (template.on_remove_script != "") {
                        CallBuffScript(template.on_remove_script, GetEntityFromBuff(bl, i), bl.buffs[i].casterEntity, bl.buffs[i].buffId)
                    }
                    bl.buffs |> erase(i)
                    bl.dirty = true
                    continue
                }
            }

            // 2. 周期性触发
            if (bl.buffs[i].period > 0.0) {
                bl.buffs[i].timeSinceLastTick += dt
                while (bl.buffs[i].timeSinceLastTick >= bl.buffs[i].period) {
                    bl.buffs[i].timeSinceLastTick -= bl.buffs[i].period

                    // 执行周期性脚本（如 DoT 伤害、HoT 治疗）
                    if (template.on_tick_script != "") {
                        CallBuffScript(template.on_tick_script, GetEntityFromBuff(bl, i), bl.buffs[i].casterEntity, bl.buffs[i].buffId)
                    }
                }
            }
        }
    }
}

4. Modifier 应用——属性计算中的 Buff 加成

// 在属性重算 (SystemRecalcStats) 中调用
def ApplyBuffModifiersToStats(var stats : BattleStats; buffList : BuffList)
{
    // 如果 dirty，重新收集所有 Modifier 并排序
    if (buffList.dirty) {
        buffList.cachedMods |> clear()
        for inst in buffList.buffs {
            for mod in inst.modifiers {
                // 层数叠加：每层追加一次效果
                for s in range(inst.currentStacks) {
                    buffList.cachedMods |> push(mod)
                }
            }
        }
        // 排序：ADD 先于 MUL，保证计算确定性
        buffList.cachedMods |> sort($(a, b) => int(a.op) < int(b.op))
        buffList.dirty = false
    }

    // 应用所有 Modifier 到 stats
    for mod in buffList.cachedMods {
        switch (mod.stat)
        {
        STAT_ATTACK:
            if (mod.op == MOD_ADD) { stats.attack += int(mod.value) }
            elif (mod.op == MOD_MUL) { stats.attack = int(float(stats.attack) * (1.0 + mod.value / 10000.0)) }
            elif (mod.op == MOD_SET_MIN) { stats.attack = max(stats.attack, int(mod.value)) }
        STAT_DEFENSE:
            if (mod.op == MOD_ADD) { stats.defense += int(mod.value) }
            elif (mod.op == MOD_MUL) { stats.defense = int(float(stats.defense) * (1.0 + mod.value / 10000.0)) }
        // ... 其余属性同理
        }
    }
}

4.1 计算确定性保证

🧠 为什么排序 Modifier：如果 Buff A 加 50 攻击（ADD）+ Buff B 乘 1.2 攻击（MUL）， 不排序时结果取决于 A 和 B 的添加顺序（50×1.2=60 vs (base+50)×1.2=base×1.2+60）。 按 MOD_ADD 先于 MOD_MUL 排序后，结果恒定：(base + ΣADD) × Π(1 + MUL/10000)。 这保证了相同的 Buff 组合在不同时间施加得出的属性结果完全一致。

5. 事件驱动的 Buff 触发

// 战斗系统中——造成伤害后检查触发型 Buff
def OnDamageDealt(attacker : Entity; defender : Entity; damage : int)
{
    ecs_edit(attacker) $(var cmp)
    {
        for inst in cmp.buffList.buffs {
            let template = g_buffTemplates.Find(inst.buffId)
            if (template != null && template.triggers |> contains(int(TRIGGER_ON_DAMAGE_DEALT))) {
                if (template.on_tick_script != "") {
                    CallBuffScriptWithContext(template.on_tick_script, attacker, defender, damage)
                }
            }
        }
    }
}

6. Buff 叠加规则

规则行为用途
Stack同 ID Buff 叠加层数，刷新持续时间DOT 堆叠（每层独立计算伤害）
Refresh同 ID Buff 只刷新持续时间，不叠加层数Buff 补持续（如坦克的防御 Buff）
Replace新 Buff 替换旧 Buff（旧立即失效）强控技能（新晕眩替换旧晕眩）
Independent多个同 ID Buff 可以共存多来源 HOT（两个治疗分别加的血量恢复）

8. Buff 互斥与免疫

// Buff 模板扩展——互斥组和免疫标签
CREATE TABLE buff_templates (
    // ... 原有字段 ...
    exclusive_group INT DEFAULT 0,       // 互斥组 ID，同组内只允许存在一个。0 = 不互斥
    immunity_tags   INT[] DEFAULT '{}'    // 拥有此 Buff 时免疫的 Buff 类型标签
);

// ApplyBuff 前置检查
def CheckBuffApplicable(target : Entity; template : BuffTemplate) : bool
{
    var applicable = true
    ecs_edit(target) $(var cmp)
    {
        // 互斥组检查
        if (template.exclusive_group != 0) {
            for inst in cmp.buffList.buffs {
                let t = g_buffTemplates.Find(inst.buffId)
                if (t != null && t.exclusive_group == template.exclusive_group) {
                    applicable = false
                    break
                }
            }
        }

        // 免疫标签检查
        for inst in cmp.buffList.buffs {
            let t = g_buffTemplates.Find(inst.buffId)
            if (t != null) {
                for tag in t.immunity_tags {
                    if (template.buff_type == tag) {
                        applicable = false
                        break break
                    }
                }
            }
        }
    }
    return applicable
}

9. 性能分析

操作频率复杂度说明
施加 Buff技能命中时（10-50/sec）O(B) B=已有Buff数遍历检查重复；通常 ≤10 个 Buff
移除 Buff到期/驱散时O(B)erase(i) 触发了 swap-with-last？不对——array erase 是 O(B)。但 B 很小
Buff 计时 Tick每 Tick 遍历所有带 Buff 的实体O(E×B)E=战斗中的实体数(≈100)，B=平均Buff数(≈5)，总计 ~500 次操作
Modifier 重收集仅 BuffList.dirty 时O(B×M)M=每个 Buff 的 Modifier 数(≈5)，排序 O(B×M log B×M)
