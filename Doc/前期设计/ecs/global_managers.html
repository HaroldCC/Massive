全局 Manager 单例

替代原先 GlobalRegistry（EnTT registry）的设计。每个跨场景子系统独立为一个 C++ 单例 Manager， 内部使用最简单的容器（unordered_map 或 vector），不引入 ECS 开销。

Mgr_1: GuildMgr — 公会管理

class GuildMgr {
public:
    static GuildMgr& Instance();

    // 生命周期
    uint32_t CreateGuild(const std::string& name, uint32_t founderId);
    void     DisbandGuild(uint32_t guildId);
    const Guild* FindByName(const std::string& name);

    // 成员管理
    void AddMember(uint32_t guildId, uint32_t playerId);
    void RemoveMember(uint32_t guildId, uint32_t playerId);
    const std::vector<uint32_t>& GetMembers(uint32_t guildId);

    // 公会仓库（第一阶段只保留占位，后续扩展）
    const GuildBank* GetBank(uint32_t guildId);

    // 持久化
    void SaveGuild(uint32_t guildId);
    void LoadAllGuilds(DbService& db);

private:
    std::unordered_map<uint32_t, Guild> _guilds;        // guildId → 公会对象
    std::unordered_map<std::string, uint32_t> _nameToId;  // 公会名 → guildId
    std::unordered_map<uint32_t, uint32_t> _playerToGuild; // playerId → guildId
};

struct Guild {
    uint32_t id;
    std::string name;
    uint32_t leaderId;
    std::vector<uint32_t> members;
    GuildBank bank;
    uint32_t level = 1;
    uint32_t experience = 0;
};

Mgr_2: ChatChannelMgr — 聊天频道管理

enum class ChatChannelType { World, Guild, Party, Private, System };

class ChatChannelMgr {
public:
    static ChatChannelMgr& Instance();

    // 发送消息——路由到正确的频道 + 在线玩家
    void SendMessage(ChatChannelType type, uint32_t senderId,
                     const std::string& message, uint32_t targetId = 0);

    // 订阅/取消
    void Subscribe(uint32_t playerId, ChatChannelType type, uint32_t channelId = 0);
    void Unsubscribe(uint32_t playerId, ChatChannelType type, uint32_t channelId = 0);

private:
    // 频道ID → 订阅者列表
    std::unordered_map<uint32_t, std::vector<uint32_t>> _subscribers;

    uint32_t ChannelKey(ChatChannelType type, uint32_t channelId) {
        return (uint32_t(type) << 24) | (channelId & 0xFFFFFF);
    }
};

Mgr_3: BossScheduler — 世界 Boss 调度

enum class BossState { Pending, Alive, Dead };

struct BossEntry {
    uint32_t bossTemplateId;
    BossState state = BossState::Pending;
    std::chrono::steady_clock::time_point nextSpawnTime;
    uint32_t sceneId = 0;           // Boss 所在的场景 ID
    Entity bossEntity;               // Boss 在 Scene 中的 Entity
    std::vector<Participant> participants; // 参与战斗的玩家（伤害/DPS 等）
};

class BossScheduler {
public:
    static BossScheduler& Instance();

    // 每 Tick 检查 Boss 刷新，由逻辑线程调用
    void Update();

    // Boss 死亡——调度下次刷新时间
    void OnBossDefeated(uint32_t bossTemplateId);

    // Boss 血量变化（转发到所有参战玩家）
    void OnBossHpChanged(uint32_t bossTemplateId, int current, int maximum);

private:
    std::unordered_map<uint32_t, BossEntry> _bosses;  // bossTemplateId → 状态
};

Mgr_4: MailMgr — 邮件系统

struct MailMessage {
    uint32_t mailId;      // 全局唯一邮件 ID
    uint32_t senderId;    // 发送者 PlayerId（0 = 系统）
    uint32_t recipientId;
    std::string title;
    std::string body;
    std::vector<ItemAttachment> attachments;
    bool read = false;
    bool claimed = false;      // 附件是否已领取
    std::chrono::steady_clock::time_point sentTime;
    std::chrono::steady_clock::time_point expireTime;  // 30 天后系统自动删除
};

class MailMgr {
public:
    static MailMgr& Instance();

    void    SendMail(const MailMessage& mail);
    void    ClaimAttachment(uint32_t mailId, uint32_t playerId);
    void    DeleteMail(uint32_t mailId);
    auto    GetInbox(uint32_t playerId) -> std::vector<const MailMessage*>;
    void    CheckExpiredMails();  // 每 5 分钟 Tick 清理过期邮件

private:
    std::vector<MailMessage> _mails;                        // 所有邮件
    std::unordered_map<uint32_t, std::vector<uint32_t>> _inboxIndex;  // playerId → mailId 列表
    std::atomic<uint32_t> _nextMailId{1};
};

Mgr_5: ConfigTable<T> — 配置表

template<typename T>
class ConfigTable {
public:
    void Load(DbService& db, const std::string& tableName);
    void Reload(DbService& db);   // 热重载：加载到 temp shared_ptr → 原子 swap
    const T* Find(uint32_t templateId) const;
    const std::vector<T>& All() const;  // 遍历（运营工具用）

private:
    std::shared_ptr<std::vector<T>> _data;               // 紧凑数组，按 templateId 排序
    std::shared_ptr<std::unordered_map<uint32_t, uint32_t>> _index;  // templateId → 数组索引
};

// 全局实例——每个模板表一个
ConfigTable<SkillTemplate>   g_skillTemplates;
ConfigTable<ItemTemplate>    g_itemTemplates;
ConfigTable<MonsterTemplate> g_monsterTemplates;
ConfigTable<QuestTemplate>   g_questTemplates;
ConfigTable<EventTemplate>   g_eventTemplates;     // 活动/节日配置
ConfigTable<MapTemplate>     g_mapTemplates;       // 副本/战场模板
