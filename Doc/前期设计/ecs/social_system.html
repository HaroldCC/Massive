社交系统 · Chat + Friend + Guild + Mail

1. 架构定位

社交系统作为 独立的 SocialServer 进程 运行。WorldServer 通过内部 TCP RPC 调用 SocialServer 接口。 GuildMgr、ChatChannelMgr、MailMgr 作为 SocialServer 进程内的单例（见 global_managers.html）。 玩家数据的社交属性（好友列表、公会 ID）通过 SocialComponent（BlobStorage）挂载在每个玩家 Entity 上（WorldServer 侧）， SocialServer 侧维护全局数据（公会表、邮件箱、拍卖行列表）。

📌 进程分离：SocialServer 是 5 种进程之一。WorldServer 通过 SocialClient（类似 CenterClient） 发起 RPC 调用。同一个 WorldServer 内的聊天频道由 ChatChannelMgr 在 SocialServer 侧管理——World 只持有一个 SocialClient 连接， 所有频道操作通过 RPC 路由到 SocialServer。

A. 聊天系统

A.1 频道类型

频道可见范围存储说明
世界 (World)同一 WorldServer 所有玩家无持久化有发送间隔限制（3s）
场景 (Zone)同一 Scene 内所有玩家无持久化/say 白字
公会 (Guild)同一公会所有在线成员无持久化仅公会成员
队伍 (Party)同一队伍成员无持久化跨场景
私聊 (Whisper)两个玩家之间无持久化跨场景
系统 (System)指定玩家/全服无持久化服务器推送公告

A.3 聊天消息处理（DasLang 侧）

// chat_handler.das — 聊天消息入口
def HandleChatSay(player : Entity; channel : int; msg : string; targetName : string = "")
{
    // 1. 脏字过滤
    let filtered = FilterChatMessage(msg)
    if (filtered == "") return

    // 2. 禁言检查
    if (IsPlayerMuted(player)) {
        SendSystemMsg(player, "你已被禁言，剩余时间: {}s", GetMuteRemaining(player))
        return
    }

    // 3. 频率限制
    if (!ChatChannelMgr.CheckRateLimit(player, ChatChannel(channel))) {
        SendSystemMsg(player, "发言太频繁，请稍后再试")
        return
    }

    // 4. 路由到对应频道
    var recipients : array<Entity>
    switch (channel)
    {
    CHAT_WORLD:
        recipients = ChatChannelMgr.SendWorld(player, filtered)
    CHAT_ZONE:
        recipients = ChatChannelMgr.SendScene(GetSceneId(player), player, filtered)
    CHAT_GUILD:
        let guildId = GetPlayerGuildId(player)
        if (guildId == 0) { SendSystemMsg(player, "你不在公会中"); return }
        recipients = ChatChannelMgr.SendGuild(guildId, player, filtered)
    CHAT_PARTY:
        let partyId = GetPlayerPartyId(player)
        if (partyId == 0) { SendSystemMsg(player, "你不在队伍中"); return }
        recipients = ChatChannelMgr.SendParty(partyId, player, filtered)
    CHAT_WHISPER:
        if (!ChatChannelMgr.SendWhisper(player, targetName, filtered)) {
            SendSystemMsg(player, "玩家 {} 不在线或不存在", targetName)
        }
        return  // 私聊走单独逻辑
    }

    // 5. 广播 MSG_CHAT_SAY_NTF 给所有接收者
    let ntf = ChatSayNtf{
        senderEntityId = player,
        senderName     = GetPlayerName(player),
        channel        = channel,
        message        = filtered
    }
    for r in recipients {
        SendToClient(GetSessionId(r), MSG_CHAT_SAY_NTF, SerializeProto(ntf))
    }
}

B.2 好友操作

// 发送好友申请 MSG_FRIEND_ADD_REQ(targetPlayerId)
def HandleFriendAdd(player : Entity; targetPlayerId : uint32)
{
    ecs_edit(player) $(var cmp)
    {
        if (cmp.friendList.entries |> length() >= cmp.friendList.maxFriends) {
            SendFriendRsp(player, FRIEND_ERR_LIST_FULL); return
        }
    }

    let target = GetEntityByPlayerId(targetPlayerId)
    if (target == InvalidEntity) { SendFriendRsp(player, FRIEND_ERR_OFFLINE); return }

    // 加入目标的待处理列表
    ecs_edit(target) $(var tCmp)
    {
        if (tCmp.friendList.pendingReqs |> contains(GetPlayerId(player))) {
            SendFriendRsp(player, FRIEND_ERR_ALREADY_PENDING); return
        }
        tCmp.friendList.pendingReqs |> push(GetPlayerId(player))
    }

    // 通知目标
    let ntf = FriendAddNtf{fromPlayerId = GetPlayerId(player), fromName = GetPlayerName(player)}
    SendToClient(GetSessionId(target), MSG_FRIEND_ADD_NTF, SerializeProto(ntf))
}

// 接受好友 MSG_FRIEND_ACCEPT_REQ(fromPlayerId)
def HandleFriendAccept(player : Entity; fromPlayerId : uint32)
{
    ecs_edit(player) $(var cmp)
    {
        // 从 pending 中移除
        RemoveFromArray(cmp.friendList.pendingReqs, fromPlayerId)

        if (cmp.friendList.entries |> length() >= cmp.friendList.maxFriends) {
            SendFriendRsp(player, FRIEND_ERR_LIST_FULL); return
        }

        cmp.friendList.entries |> push(FriendEntry{playerId = fromPlayerId, addedTime = GetTickTime(), note = ""})
    }

    let fromEntity = GetEntityByPlayerId(fromPlayerId)
    if (fromEntity != InvalidEntity) {
        ecs_edit(fromEntity) $(var fCmp) {
            fCmp.friendList.entries |> push(FriendEntry{playerId = GetPlayerId(player), addedTime = GetTickTime(), note = ""})
        }
    }

    // 立即写 DB——LINQ 管道链（编译期为 SQL，无手写 SQL 字符串）
    DBInsert("player_friends", [
        { player_id = GetPlayerId(player), friend_id = fromPlayerId },
        { player_id = fromPlayerId, friend_id = GetPlayerId(player) }
    ])
}

C.2 公会操作（DasLang 侧）

// guild_handler.das — 公会消息入口
def HandleGuildCreate(player : Entity; guildName : string)
{
    // 校验
    if (GetPlayerLevel(player) < 20) { SendGuildRsp(player, GUILD_ERR_LEVEL_TOO_LOW); return }
    if (GetPlayerGuildId(player) != 0) { SendGuildRsp(player, GUILD_ERR_ALREADY_IN_GUILD); return }
    if (guildName |> length() < 2 || guildName |> length() > 12) {
        SendGuildRsp(player, GUILD_ERR_INVALID_NAME); return
    }

    let gold = GetPlayerGold(player)
    if (gold < kGuildCreateCost) { SendGuildRsp(player, GUILD_ERR_NOT_ENOUGH_GOLD); return }

    let guildId = GuildMgr.CreateGuild(guildName, GetPlayerId(player), GetPlayerName(player))
    if (guildId == 0) { SendGuildRsp(player, GUILD_ERR_NAME_TAKEN); return }

    DeductGold(player, kGuildCreateCost)
    SetPlayerGuildId(player, guildId)
    ChatChannelMgr.JoinGuild(guildId, player)

    SendGuildRsp(player, GUILD_OK)
    DBInsert("guilds", [{ guild_id = guildId, guild_name = guildName, creator_id = GetPlayerId(player) }])

    LOG_INFO(0, "Guild '{}' ({}) created by player {}", guildName, guildId, GetPlayerId(player))
}

def HandleGuildDisband(player : Entity)
{
    let guildId = GetPlayerGuildId(player)
    if (guildId == 0) { SendGuildRsp(player, GUILD_ERR_NOT_IN_GUILD); return }

    let guild = GuildMgr.GetGuild(guildId)
    if (guild == null || guild.creatorId != GetPlayerId(player)) {
        SendGuildRsp(player, GUILD_ERR_NOT_LEADER); return
    }

    // 通知所有在线成员
    let onlineMembers = GuildMgr.GetOnlineMembers(guildId)
    for memberId in onlineMembers {
        let member = GetEntityByPlayerId(memberId)
        if (member != InvalidEntity) {
            SetPlayerGuildId(member, 0)
            ChatChannelMgr.LeaveGuild(guildId, member)
        }
    }

    GuildMgr.DisbandGuild(guildId, GetPlayerId(player))
    table("guilds") |> where_($(r) => r.guild_id == guildId) |> DBDelete()
    for memberId in onlineMembers {
        SendToClient(GetSessionId(GetEntityByPlayerId(memberId)), MSG_GUILD_DISBAND_NTF, "")
    }
}

D.3 邮件入口（DasLang 侧）

def HandleMailClaim(player : Entity; mailId : uint64)
{
    var items : array<Attachment>
    let ok = MailMgr.ClaimAttachment(GetPlayerId(player), mailId, items)
    if (!ok) { SendMailClaimRsp(player, MAIL_ERR_ALREADY_CLAIMED); return }

    // 检查背包空间
    ecs_edit(player) $(var cmp)
    {
        let spaceNeeded = items |> length()
        let spaceAvailable = CountEmptyBagSlots(cmp.inventory)
        if (spaceAvailable < spaceNeeded) {
            // 空间不足——回滚领取状态（或保持已领取但物品留邮件中）
            return SendMailClaimRsp(player, MAIL_ERR_BAG_FULL)
        }

        for att in items {
            let item = CreateItemInstance(att.itemId, att.count)
            let emptySlot = FindEmptyBagSlot(cmp.inventory)
            cmp.inventory.slots[emptySlot].item = item
        }
    }

    MarkInventoryDirty(player)
    SendMailClaimRsp(player, MAIL_OK)
}
