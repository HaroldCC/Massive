AI 框架 · 行为树 + 黑板 + 日程调度

1. 设计概述

AI 系统负责所有非玩家实体（怪物、NPC、宠物、召唤物）的自主决策。采用 行为树（Behavior Tree） 做决策框架， 黑板（Blackboard） 做状态存储。行为树节点在 DasLang 中定义（可热更新）， Tick 由 C++ 调度器以固定频率驱动。支持分帧执行——一个复杂 AI 可分多 Tick 完成决策，保证 Tick 预算可控。

2. 架构分层

层级职责实现方
调度器管理 AI Tick 频率、分帧、实体分组、LODC++
行为树引擎节点 Tick/Reset、子树、并行节点、装饰器C++（核心）+ DasLang（节点注册）
行为树节点具体条件和动作（CheckDistance, MoveTo, CastSkill...）DasLang
黑板AI 状态存储（目标、仇恨、巡逻路径索引...）ScriptComponent（BlobStorage）

3. 黑板（Blackboard）——ScriptComponent

// AI 黑板——每个 AI 实体各自独立的 ScriptComponent
[script_component]
struct AIBlackboard {
    // 目标管理
    currentTarget     : Entity              // 当前战斗目标
    threatTarget      : Entity              // 仇恨最高目标

    // 状态机状态
    aiState           : int                 // 0=Idle 1=Patrol 2=Combat 3=Flee 4=Return 5=Dead
    lastStateChange   : float               // TickTime

    // 巡逻
    patrolPathId      : int                 // 巡逻路径 ID（0=无巡逻）
    patrolIndex       : int                 // 当前巡逻点索引
    patrolDirection   : int                 // 1=正序 -1=逆序
    patrolWaitUntil   : float               // 巡逻点等待到何时

    // 出生点
    spawnPosition     : Vector3             // 出生/返回位置
    chaseRange        : float               // 追击范围（从模板初始化）
    leashRange        : float               // 脱战范围（超出则回出生点）

    // 技能选择
    availableSkillIds : array<uint32>        // 可用技能列表
    lastSkillCastTime : float               // 上次施法时间

    // 临时数据（行为树节点间共享）
    tempFloats        : array<float>         // 临时浮点值
    tempEntities      : array<Entity>        // 临时实体引用
    tempInts          : array<int>

    // 行为树运行时
    btStack           : array<int>           // 行为树调用栈（用于分帧恢复）
}

4. 行为树节点类型

类型返回值说明
Sequence（顺序）依次执行子节点，任一失败则失败"且"逻辑——所有条件满足才继续
Selector（选择）依次执行子节点，任一成功则成功"或"逻辑——优先选择第一个可执行的分支
Parallel（并行）同时执行所有子节点移动+攻击 同时进行
Condition（条件）检查条件，成功→继续，失败→终止距离检查、HP检查、Buff检查
Action（动作）执行动作，返回 Running/Success/Fail移动、施法、巡逻、逃跑
Decorator（装饰器）包装子节点，修改其行为Inverter、Repeater、Timeout、Cooldown

5. 行为树节点——DasLang 实现

// ai_nodes.das — 基础行为树节点

enum BTStatus { BT_SUCCESS, BT_FAIL, BT_RUNNING }

// ==== 条件节点 ====

// 检查是否有仇恨目标
def BTCond_HasThreatTarget(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        cmp.aiBlackboard.threatTarget = GetTopThreatTarget(entity)
        return cmp.aiBlackboard.threatTarget != InvalidEntity ? BT_SUCCESS : BT_FAIL
    }
    return BT_FAIL
}

// 检查目标是否在近战范围内
def BTCond_InMeleeRange(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let target = cmp.aiBlackboard.currentTarget
        if (target == InvalidEntity) return BT_FAIL
        let dist = DistanceBetween(entity, target)
        let meleeRange = GetMeleeRange(entity)  // 从 MonsterTemplate 查
        return dist <= meleeRange ? BT_SUCCESS : BT_FAIL
    }
    return BT_FAIL
}

// 检查目标是否在追击范围内
def BTCond_InChaseRange(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let target = cmp.aiBlackboard.currentTarget
        if (target == InvalidEntity) return BT_FAIL
        let dist = DistanceBetween(entity, target)
        return dist <= cmp.aiBlackboard.chaseRange ? BT_SUCCESS : BT_FAIL
    }
    return BT_FAIL
}

// 检查 HP 低于阈值
def BTCond_HpBelow(entity : Entity; thresholdPct : int) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let pct = cmp.battleStats.currentHp * 100 / cmp.battleStats.maxHp
        return pct < thresholdPct ? BT_SUCCESS : BT_FAIL
    }
    return BT_FAIL
}

// ==== 动作节点 ====

// 向目标移动
def BTAction_MoveToTarget(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let target = cmp.aiBlackboard.currentTarget
        if (target == InvalidEntity) return BT_FAIL

        // 寻路到目标位置
        let path = NavFindPath(GetSceneId(entity), GetPosition(entity), GetPosition(target))
        if (path |> empty()) return BT_FAIL

        // 向路径第一个点移动
        MoveEntityToward(entity, path[0], GetMoveSpeed(entity))
        return BT_RUNNING
    }
    return BT_FAIL
}

// 攻击目标（选择最优技能）
def BTAction_AttackTarget(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let target = cmp.aiBlackboard.currentTarget
        if (target == InvalidEntity) return BT_FAIL

        // 选择可用技能（不在 CD 中、有资源的）
        let skillId = SelectBestSkill(entity, cmp.aiBlackboard.availableSkillIds)
        if (skillId == 0) {
            // 没有可用技能——普攻
            return BTAction_AutoAttack(entity)
        }

        let result = HandleSkillCast(entity, skillId, target, 0.0, 0.0, 0.0)
        return result == SKILL_OK ? BT_SUCCESS : BT_FAIL
    }
    return BT_FAIL
}

// 巡逻——沿预设路径移动到下一个巡逻点
def BTAction_Patrol(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let bb = cmp.aiBlackboard

        // 检查是否正在等待
        if (bb.patrolWaitUntil > GetTickTime()) return BT_RUNNING

        // 获取当前巡逻点
        let waypoint = GetPatrolWaypoint(bb.patrolPathId, bb.patrolIndex)
        let dist = DistanceToPosition(entity, waypoint)

        if (dist < 1.0) {
            // 到达巡逻点——等待 2 秒，然后前进
            bb.patrolWaitUntil = GetTickTime() + 2.0
            bb.patrolIndex = GetNextPatrolIndex(bb)
        }

        // 向巡逻点移动
        MoveEntityTowardPosition(entity, waypoint, GetWalkSpeed(entity))
        return BT_RUNNING
    }
    return BT_FAIL
}

// 返回出生点
def BTAction_ReturnToSpawn(entity : Entity) : BTStatus
{
    ecs_edit(entity) $(var cmp)
    {
        let dist = DistanceToPosition(entity, cmp.aiBlackboard.spawnPosition)
        if (dist < 1.0) {
            // 到达出生点——重置 AI 状态
            cmp.aiBlackboard.aiState = AI_STATE_IDLE
            cmp.aiBlackboard.currentTarget = InvalidEntity
            ClearThreatTable(entity)
            return BT_SUCCESS
        }

        // 移向出生点
        MoveEntityTowardPosition(entity, cmp.aiBlackboard.spawnPosition, GetMoveSpeed(entity))
        return BT_RUNNING
    }
    return BT_FAIL
}

6. 行为树组装——通用怪物 AI 模板

// 怪物通用行为树（Selector 根节点）
//
// Root (Selector)
//   ├── Sequence: 战斗
//   │     ├── Cond: 有仇恨目标
//   │     ├── Cond: 目标在追击范围内
//   │     ├── Selector: 战斗行为
//   │     │     ├── Sequence: 近战攻击
//   │     │     │     ├── Cond: 在近战范围内
//   │     │     │     └── Action: 攻击目标
//   │     │     └── Action: 移向目标
//   │     └── Action: 更新战斗状态
//   ├── Sequence: 返回出生点
//   │     ├── Cond: AI 状态为 "返回"
//   │     └── Action: 返回出生点
//   ├── Sequence: 巡逻
//   │     ├── Cond: 有巡逻路径
//   │     └── Action: 巡逻
//   └── Action: 待机（播放 Idle 动画）

// 行为树运行——DasLang 侧调度
def RunBehaviorTree(entity : Entity) : BTStatus
{
    // Root Selector: 按优先级尝试各分支

    // 分支 1: 战斗
    if (BTCond_HasThreatTarget(entity) == BT_SUCCESS) {
        ecs_edit(entity) $(var cmp) {
            cmp.aiBlackboard.currentTarget = cmp.aiBlackboard.threatTarget
            cmp.aiBlackboard.aiState = AI_STATE_COMBAT
        }

        if (BTCond_InChaseRange(entity) == BT_SUCCESS) {
            // 检查是否需要逃跑
            if (ShouldFlee(entity)) {
                FleeFromTarget(entity)
                return BT_RUNNING
            }

            if (BTCond_InMeleeRange(entity) == BT_SUCCESS) {
                BTAction_AttackTarget(entity)
            } else {
                BTAction_MoveToTarget(entity)
            }
            return BT_RUNNING
        } else {
            // 超出追击范围——脱战，返回出生点
            ecs_edit(entity) $(var cmp) {
                cmp.aiBlackboard.aiState = AI_STATE_RETURN
                cmp.aiBlackboard.currentTarget = InvalidEntity
            }
            ClearThreatTable(entity)
            return BT_RUNNING
        }
    }

    // 分支 2: 返回出生点
    if (GetAIState(entity) == AI_STATE_RETURN) {
        return BTAction_ReturnToSpawn(entity)
    }

    // 分支 3: 巡逻
    if (HasPatrolPath(entity)) {
        if (GetAIState(entity) != AI_STATE_PATROL) {
            SetAIState(entity, AI_STATE_PATROL)
        }
        return BTAction_Patrol(entity)
    }

    // 分支 4: 待机
    return BT_SUCCESS
}

7. AI 调度器（C++ 侧）

// AIScheduler.h — AI 调度器（C++ 引擎层）
class AIScheduler {
public:
    struct Config {
        int aiTickIntervalMs = 200;      // AI 决策间隔：200ms（每 4 个逻辑 Tick）
        int maxAiPerTick      = 500;      // 每逻辑 Tick 最多处理 500 个 AI 实体
        int aiBatchSize       = 50;       // 分帧——每 Tick 处理 50 个 AI 决策
        int maxBTDepth        = 32;       // 行为树最大递归深度（防栈溢出）
    };

    // 每 Tick 由逻辑线程调用
    void Tick() {
        // Round-Robin 遍历所有 AI 实体，每个实体每 200ms 决策一次
        if (_batchRemaining <= 0) return;  // 本 Tick 预算用尽

        int processed = 0;
        while (_cursor != _aiEntities.end() && processed < _batchRemaining) {
            auto entity = *_cursor;
            ++_cursor;

            // 检查是否到该实体的 AI Tick 时间
            auto& lastTick = _lastAiTick[entity];
            if (_tickTime - lastTick >= _config.aiTickIntervalMs) {
                lastTick = _tickTime;
                RunEntityAI(entity);  // → DasLang RunBehaviorTree
                processed++;
            }
        }

        // 游标回到起点——Round-Robin 公平调度
        if (_cursor == _aiEntities.end())
            _cursor = _aiEntities.begin();
    }

    void RegisterEntity(Entity entity) { _aiEntities.push_back(entity); }
    void UnregisterEntity(Entity entity) { /* 标记删除，下 Tick 清理 */ }

private:
    Config _config;
    std::vector<Entity> _aiEntities;
    std::vector<Entity>::iterator _cursor = _aiEntities.begin();
    std::unordered_map<Entity, std::chrono::milliseconds> _lastAiTick;
    int _batchRemaining = _config.aiBatchSize;
    std::chrono::milliseconds _tickTime;
};

7.1 行为树深度保护

行为树节点的递归执行可能导致栈溢出——尤其是 DasLang 的实现栈小于 C++ 原生栈。 AI 调度器在每次 RunEntityAI 调用前重置 depth 计数器， DasLang 侧的 BT 节点每次递归子节点时递增。超过硬上限（32 层）时截断并告警。 

// C++ 层在调用 DasLang 前设置深度限制
static constexpr int kMaxBTDepth = 32;

NodeStatus EvaluateTree(BTNode* node, AIBlackboard& bb, int depth = 0) {
    if (depth > kMaxBTDepth) {
        LOG_ERROR("BT depth exceeded {} for entity", kMaxBTDepth);
        TracyMessageL("BT depth overflow");
        return NodeStatus::Failure;  // 截断——静默失败，不崩进程
    }

    switch (node->type) {
        case NodeType::Selector:
            for (auto& child : node->children) {
                auto status = EvaluateTree(child, bb, depth + 1);
                if (status == NodeStatus::Success) return NodeStatus::Success;
            }
            return NodeStatus::Failure;

        case NodeType::Sequence:
            for (auto& child : node->children) {
                auto status = EvaluateTree(child, bb, depth + 1);
                if (status != NodeStatus::Success) return status;
            }
            return NodeStatus::Success;

        default:
            return node->Execute(bb);  // 叶子节点——DasLang 实现
    }
}
深度限制 32 层：你的怪物通用 BT 模板只有 5 层深（Selector → Sequence → Selector → Sequence → Action）。 32 层对复杂 Boss AI 也绰绰有余——超过这个深度通常意味着设计错误而非需要更深的树。 截断时打 ERROR 日志 + Tracy 标记，立即引起注意。

8. AI LOD（细节层次）

LOD距离范围AI 频率行为
LOD 0（近）0-50m每 200ms完整行为树：战斗/巡逻/技能选择
LOD 1（中）50-150m每 1s简化：只更新移动和状态机
LOD 2（远）150m+每 5s极简：只检查是否需要进入 LOD 1
// LOD 计算——基于与最近玩家的距离
def GetAILOD(entity : Entity) : int
{
    let nearestPlayerDist = GetDistanceToNearestPlayer(entity)
    if (nearestPlayerDist < 50.0)  return 0
    if (nearestPlayerDist < 150.0) return 1
    return 2
}

9. NPC 日程系统（Schedule System）

城镇 NPC 不跑行为树，跑 日程表——按时间切换位置和行为。比行为树更轻量。 

// 脚本组件——NPC 日程
[script_component]
struct NPCSchedule {
    entries     : array<NPCScheduleEntry>
}

struct NPCScheduleEntry {
    startHour   : int           // 游戏内时间（小时）
    endHour     : int
    sceneId     : uint32        // 所在的场景
    position    : Vector3
    animation   : int           // 播放动画 ID（0=默认待机）
    dialogueId  : int           // 可交互的对话 ID（0=不可交互）
}

// NPC AI Tick——简单日程切换
def RunNPCSchedule(entity : Entity)
{
    let gameHour = GetGameTimeHour()
    ecs_edit(entity) $(var cmp)
    {
        for entry in cmp.npcSchedule.entries {
            if (gameHour >= entry.startHour && gameHour < entry.endHour) {
                // 检查是否需要切换
                if (cmp.npcSchedule.currentEntry != entry) {
                    // 跨场景移动（如从广场走入屋内）
                    if (entry.sceneId != GetSceneId(entity)) {
                        MigrateEntityToScene(entity, entry.sceneId)
                    }
                    TeleportEntity(entity, entry.position)
                    SetAnimation(entity, entry.animation)
                    SetDialogue(entity, entry.dialogueId)
                    cmp.npcSchedule.currentEntry = entry
                }
                return
            }
        }
    }
}
