技能系统 · 施法管线 + 冷却管理 + 目标选择

1. 设计概述

技能系统负责技能施放的完整生命周期：客户端请求 → 服务器校验 → 施法 → 效果结算 → 冷却管理。 采用 固定阶段流水线（同伤害管线），无虚函数、零堆分配。技能数据从 ConfigTable 查表， 施法逻辑在 DasLang 脚本中实现（可热更新）。

2. 技能模板（ConfigTable）

CREATE TABLE skill_templates (

    gcd_category      INT DEFAULT 0,         // GCD 分组（0=独立，1=通用GCD）
    gcd_ms            INT DEFAULT 1500,       // GCD 时间（毫秒）
    cost_mp           INT DEFAULT 0,         // MP 消耗
    cost_hp           INT DEFAULT 0,         // HP 消耗
    cost_rage         INT DEFAULT 0,         // 怒气/能量消耗
    min_level          INT DEFAULT 1,
    require_weapon     INT DEFAULT 0,         // 0=不限 1=需装备武器
    on_cast_script     TEXT,                // 施法成功后执行的脚本函数名
    on_hit_script      TEXT,                // 命中目标后执行的脚本函数名
    icon               TEXT,
    description        TEXT
);

4. 施法管线（Cast Pipeline）

4.1 管线阶段

Client → MSG_SKILL_CAST_REQ(skillId, targetEntityId, targetX, targetY, targetZ)
  │
  ├── 阶段 1: 基础校验
  │     □ caster 未死亡 (no DeadTag)
  │     □ caster 未被控制 (no StunnedTag/SilencedTag/DisarmedTag)
  │     □ caster 不在读条中 (castState == IDLE)
  │     □ 技能在冷却中？→ REJECTED "on cooldown"
  │     □ GCD 未结束？→ REJECTED "gcd active"
  │     □ caster 未进入战斗？被动恢复技能允许，其余需要 CombatTag
  │
  ├── 阶段 2: 资源校验
  │     □ currentMp ≥ cost_mp → 否则 REJECTED "not enough mp"
  │     □ currentHp ≥ cost_hp（且 currentHp - cost_hp > 0）→ 否则 REJECTED
  │     □ 其他资源（怒气/能量/连击点）检查
  │
  ├── 阶段 3: 目标校验
  │     □ target_type == SELF → 自动通过
  │     □ target_type == SINGLE → target 存在且有效、未死亡、非无敌
  │     □ target_type == AOE_POINT → (targetX, targetZ) 在 NavMesh 上
  │     □ 距离检查：Distance(caster, target) ≤ max_range → 否则 REJECTED "out of range"
  │     □ 视线检查：IsInLineOfSight(caster, target) → 否则 REJECTED
  │     □ PVP 标记检查（是否允许攻击同阵营/和平玩家）
  │
  ├── 阶段 4: 施法执行
  │     □ 瞬发(cast_time_ms=0) → 立即执行效果
  │     □ 读条(cast_time_ms>0) → 设置 CastState=CASTING + 注册定时器
  │     □ 引导(channel_time_ms>0) → 设置 CastState=CHANNELING + 周期性 Tick
  │
  ├── 阶段 5: 效果结算
  │     □ 扣资源（MP/HP/怒气）
  │     □ 设置冷却 + GCD
  │     □ 执行 on_cast_script → 生成 DamageEvent / HealEvent / BuffApply
  │     □ 执行 on_hit_script（对每个目标）
  │     □ 广播 MSG_SKILL_CAST_NTF 给附近玩家（动画同步）
  │
  └── 阶段 6: 回复
        MSG_SKILL_CAST_RSP(accepted, skillId, cooldownEnd, gcdEnd)

4.2 施法状态机

// CastState — EnTT Tag 组件
enum class CastState : uint8_t {
    Idle        = 0,   // 空闲
    Casting     = 1,   // 读条中（不可移动，不可施放其他技能）
    Channeling  = 2,   // 引导中（可移动视技能而定）
};

struct CastingComponent {
    uint32_t  skillId;
    Entity    targetEntity;
    Vector3   targetPos;    // AoE 用
    uint32_t  timerId;      // 定时器 ID（取消读条用）
    CastState state;
};

// 施法校验 + 执行——DasLang 侧
def HandleSkillCast(caster : Entity; skillId : uint32; target : Entity; targetX : float; targetY : float; targetZ : float)
{
    let template = g_skillTemplates.Find(skillId)
    if (template == null) {
        SendSkillCastRsp(caster, skillId, SKILL_ERR_NOT_FOUND)
        return
    }

    // === 阶段 1: 基础校验 ===
    var canCast = true
    var errorCode = SKILL_OK

    ecs_edit(caster) $(var cmp)
    {
        // 状态检查
        if (HasTag(caster, DeadTag) || HasTag(caster, StunnedTag)) {
            canCast = false; errorCode = SKILL_ERR_CANT_CAST
        }

        // 读条中不可施法
        if (canCast && GetCastState(caster) != CastState::Idle) {
            canCast = false; errorCode = SKILL_ERR_ALREADY_CASTING
        }

        // GCD 检查
        if (canCast && cmp.cooldowns.gcdEndTime > GetTickTime()) {
            canCast = false; errorCode = SKILL_ERR_GCD_ACTIVE
        }

        // 冷却检查
        if (canCast) {
            for cd in cmp.cooldowns.entries {
                if (cd.skillId == skillId && cd.endTime > GetTickTime()) {
                    canCast = false; errorCode = SKILL_ERR_ON_COOLDOWN
                    break
                }
            }
        }
    }

    if (!canCast) { SendSkillCastRsp(caster, skillId, errorCode); return }

    // === 阶段 2: 资源校验 ===
    ecs_edit(caster) $(var cmp)
    {
        if (template.cost_mp > cmp.battleStats.currentMp) {
            errorCode = SKILL_ERR_NOT_ENOUGH_MP
        } elif (template.cost_hp > 0 && cmp.battleStats.currentHp - template.cost_hp <= 0) {
            errorCode = SKILL_ERR_NOT_ENOUGH_HP
        }
    }
    if (errorCode != SKILL_OK) { SendSkillCastRsp(caster, skillId, errorCode); return }

    // === 阶段 3: 目标校验 ===
    if (template.target_type != SKILL_TARGET_SELF) {
        if (target == InvalidEntity || !IsValidEntity(target)) {
            SendSkillCastRsp(caster, skillId, SKILL_ERR_INVALID_TARGET); return
        }
        if (HasTag(target, DeadTag)) {
            SendSkillCastRsp(caster, skillId, SKILL_ERR_TARGET_DEAD); return
        }
        if (DistanceBetween(caster, target) > template.max_range) {
            SendSkillCastRsp(caster, skillId, SKILL_ERR_OUT_OF_RANGE); return
        }
    }

    // === 阶段 4: 扣资源 + 设冷却（先扣再执行，防止脚本出错重复扣）===
    ecs_edit(caster) $(var cmp)
    {
        cmp.battleStats.currentMp -= template.cost_mp
        cmp.battleStats.currentHp -= template.cost_hp

        // 设置冷却
        if (template.cooldown_ms > 0) {
            SetCooldown(cmp.cooldowns, skillId, GetTickTime() + float(template.cooldown_ms) / 1000.0)
        }

        // 设置 GCD
        if (template.gcd_category != 0) {
            cmp.cooldowns.gcdEndTime = GetTickTime() + float(template.gcd_ms) / 1000.0
        }
    }

    // === 阶段 5: 读条（如需要）===
    if (template.cast_time_ms > 0) {
        let timerId = ScheduleTimer(template.cast_time_ms, $(timerId: uint32) {
            ExecuteSkillEffect(caster, template, target, targetX, targetY, targetZ)
            SetCastState(caster, CastState::Idle)
        })
        SetCasting(caster, skillId, target, timerId, CastState::Casting)
    } else {
        // 瞬发——立即执行
        ExecuteSkillEffect(caster, template, target, targetX, targetY, targetZ)
    }

    // 广播施法动画
    let ntf = SkillCastNtf{
        casterEntityId = caster,
        skillId = skillId,
        targetEntityId = target,
        castTimeMs = template.cast_time_ms
    }
    BroadcastNearby(caster, MSG_SKILL_CAST_NTF, SerializeProto(ntf))

    // 回复客户端——先发 RSP 表示接受了施法请求
    SendSkillCastRsp(caster, skillId, SKILL_OK)
}

5. 效果执行——脚本驱动

// 技能效果执行——DasLang 侧
def ExecuteSkillEffect(caster : Entity; template : SkillTemplate; primaryTarget : Entity;
                       targetX : float; targetY : float; targetZ : float)
{
    // 1. 收集目标列表（单体 vs AoE）
    var targets : array<Entity>
    if (template.target_type == SKILL_TARGET_SINGLE) {
        targets |> push(primaryTarget)
    } elif (template.target_type == SKILL_TARGET_AOE_POINT) {
        targets = GetEntitiesInAoiRadius(GetSceneId(caster), targetX, targetZ, template.aoe_radius)
    } elif (template.target_type == SKILL_TARGET_AOE_SELF) {
        let pos = GetPosition(caster)
        targets = GetEntitiesInAoiRadius(GetSceneId(caster), pos.x, pos.z, template.aoe_radius)
    }

    // 2. 对每个目标执行 on_hit_script
    if (template.on_hit_script != "") {
        for t in targets {
            CallSkillScript(template.on_hit_script, caster, t, template)
        }
    }

    // 3. 执行 on_cast_script（施法者自身的额外效果）
    if (template.on_cast_script != "") {
        CallSkillScript(template.on_cast_script, caster, primaryTarget, template)
    }
}

// 示例：火球术 on_hit 脚本
def SkillScript_Fireball_Hit(caster : Entity; target : Entity; template : SkillTemplate)
{
    // 伤害结算
    let baseDamage = CalcSkillDamage(caster, template.skill_id)
    HandleSkillDamage(caster, target, template.skill_id, baseDamage)

    // 附加效果：15% 概率附加灼烧 Debuff
    if (RandomFloat(0.0, 1.0) < 0.15) {
        ApplyBuff(target, caster, BUFF_BURNING, 1)
    }
}

6. 冷却管理

// Cooldowns 操作（DasLang 侧）

def SetCooldown(var cds : Cooldowns; skillId : uint32; endTime : float)
{
    // 查找是否已有该技能的冷却记录
    for var cd in cds.entries {
        if (cd.skillId == skillId) {
            cd.endTime = endTime     // 覆盖
            return
        }
    }

    // 新记录
    cds.entries |> push(CooldownEntry{skillId = skillId, endTime = endTime})
}

def GetCooldownRemaining(cds : Cooldowns; skillId : uint32) : float
{
    for cd in cds.entries {
        if (cd.skillId == skillId) {
            return max(0.0, cd.endTime - GetTickTime())
        }
    }
    return 0.0
}

// 定期清理过期冷却记录（可在 SystemSkillCooldown 中调用）
def CleanupExpiredCooldowns(var cds : Cooldowns)
{
    let now = GetTickTime()
    for var i = cds.entries |> length() - 1; i >= 0; i-- {
        if (cds.entries[i].endTime <= now) {
            cds.entries |> erase(i)
        }
    }
}

7. GCD 管理

GCD 分组说明示例
0 (独立)不触发任何 GCD，也不受其他技能 GCD 影响药水、饰品、位移技能
1 (通用GCD)触发通用 GCD（默认 1.5s），受攻击速度影响大多数攻击技能、法术
// GCD 计算公式：实际 GCD = 基础GCD × (1 - 急速加成)
def GetActualGcdMs(caster : Entity; template : SkillTemplate) : int
{
    if (template.gcd_category == 0) return 0

    ecs_edit(caster) $(var cmp)
    {
        let haste = float(cmp.battleStats.attackSpeed) / 10000.0  // attackSpeed 是万分比
        return int(float(template.gcd_ms) / (1.0 + haste))
    }
    return template.gcd_ms
}

8. 读条打断

// 读条可被打断的场景
def InterruptCasting(caster : Entity; reason : InterruptReason)
{
    ecs_edit(caster) $(var cmp)
    {
        let castComp = GetCastingComponent(caster)
        if (castComp.state == CastState::Idle) return

        // 取消读条定时器
        CancelTimer(castComp.timerId)

        // 返还资源（或部分返还）
        let template = g_skillTemplates.Find(castComp.skillId)
        if (template != null) {
            if (reason == InterruptReason::PlayerCancelled) {
                // 主动取消——全额返还
                cmp.battleStats.currentMp += template.cost_mp
            } elif (reason == InterruptReason::Interrupted) {
                // 被打断——返还 50%
                cmp.battleStats.currentMp += template.cost_mp / 2
            }
            // InterruptReason::CancelledByMove — 移动取消，不返还
        }

        SetCastState(caster, CastState::Idle)
        SendSkillCastRsp(caster, castComp.skillId, SKILL_ERR_INTERRUPTED)
    }
}

enum InterruptReason {
    INTERRUPT_PLAYER,       // 玩家主动取消（按 ESC）
    INTERRUPT_MOVE,         // 移动打断
    INTERRUPT_STUN,         // 被控制打断
    INTERRUPT_KNOCKBACK,    // 击退打断
    INTERRUPT_COUNTER,      // 打断技能打断
}

9. 技能协议

// Proto 定义
message SkillCastReq {
    uint32 skill_id = 1;
    uint32 target_entity_id = 2;        // 单体目标（AoE 时留空 or 中心目标）
    float  target_x = 3;                // AoE 坐标
    float  target_y = 4;
    float  target_z = 5;
}

message SkillCastRsp {
    uint32 skill_id = 1;
    SkillErrorCode error_code = 2;      // 0=成功
    float  cooldown_end = 3;            // 冷却结束时刻（客户端同步用）
    float  gcd_end = 4;                 // GCD 结束时刻
}

message SkillCastNtf {
    uint32 caster_entity_id = 1;        // 施法者
    uint32 skill_id = 2;
    uint32 target_entity_id = 3;        // 目标（AoE 时为 0）
    float  target_x = 4;                // AoE 中心点
    float  target_y = 5;
    float  target_z = 6;
    int32  cast_time_ms = 7;            // 读条时间（客户端播施法动画）
}

enum SkillErrorCode {
    SKILL_OK               = 0;
    SKILL_ERR_NOT_FOUND    = 1;
    SKILL_ERR_ON_COOLDOWN  = 2;
    SKILL_ERR_GCD_ACTIVE   = 3;
    SKILL_ERR_NOT_ENOUGH_MP = 4;
    SKILL_ERR_NOT_ENOUGH_HP = 5;
    SKILL_ERR_INVALID_TARGET = 6;
    SKILL_ERR_TARGET_DEAD  = 7;
    SKILL_ERR_OUT_OF_RANGE = 8;
    SKILL_ERR_CANT_CAST    = 9;         // 晕眩/沉默/缴械等
    SKILL_ERR_ALREADY_CASTING = 10;
    SKILL_ERR_INTERRUPTED  = 11;
}
