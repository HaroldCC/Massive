匹配系统 · 副本 + 战场 + 竞技场

1. 设计概述

匹配系统负责将玩家分配到副本、战场、竞技场等实例场景中。采用 MatchQueue + MatchSolver 双组件架构： MatchQueue 管理排队玩家，MatchSolver 按规则（等级范围、MMR、队伍）组合匹配。 作为 SocialServer 内的子系统 运行——匹配逻辑与公会/好友等全局状态放在同一进程， 方便跨服组队——SocialServer 通过 CenterServer 获取各 WorldServer 玩家在线状态。

3. MatchQueue 设计（C++ 单例）

// MatchQueue.h — 匹配队列管理
class MatchQueue {
public:
    static MatchQueue& Instance() { static MatchQueue inst; return inst; }

    enum class MatchType { Dungeon, Battleground, Arena };

    struct MatchEntry {
        Entity player;               // 玩家 Entity
        uint32_t playerId;
        MatchType matchType;
        uint32_t dungeonId = 0;      // 副本 ID（副本匹配用）
        uint32_t battlegroundId = 0;  // 战场 ID
        uint32_t arenaId = 0;         // 竞技场 ID
        int level;
        int mmr;                      // 匹配分（天梯分）
        int role;                     // 0=Tank 1=Healer 2=DPS
        uint32_t partyId = 0;         // 组队 ID（0=散排）
        std::chrono::steady_clock::time_point enqueueTime;
        bool accepted = false;        // 已确认进入
    };

    // 入队
    bool Enqueue(const MatchEntry& entry);

    // 离队（取消匹配）
    void Dequeue(Entity player, MatchType matchType);

    // 确认进入（匹配成功后客户端点"进入"）
    void Accept(Entity player, MatchType matchType);

    // 获取某类型的队列
    const std::vector<MatchEntry>& GetQueue(MatchType type, uint32_t specificId = 0);

    // 移除已确认的玩家（匹配成功并确认后调用）
    void RemoveAccepted(const std::vector<Entity>& players, MatchType matchType);

    // 超时检查——每 30s 清理超时确认的玩家
    void CheckTimeouts();

private:
    // 副本队列：dungeonId → 排队玩家
    std::unordered_map<uint32_t, std::vector<MatchEntry>> _dungeonQueues;

    // 战场队列：battlegroundId → 排队玩家
    std::unordered_map<uint32_t, std::vector<MatchEntry>> _battlegroundQueues;

    // 竞技场队列：arenaId → 排队玩家
    std::unordered_map<uint32_t, std::vector<MatchEntry>> _arenaQueues;

    static constexpr auto kAcceptTimeout = std::chrono::seconds(30);
};

4. MatchSolver 匹配算法（C++ 引擎层）

4.1 副本匹配（5 人）

// MatchSolver.h — 匹配求解器
class MatchSolver {
public:
    struct MatchResult {
        std::vector<Entity> players;
        uint32_t dungeonId = 0;
        uint32_t battlegroundId = 0;
        uint32_t arenaId = 0;
        MatchQueue::MatchType type;
        int averageMmr = 0;
    };

    // 每 5 Tick (100ms) 由逻辑线程调用——尝试求解所有队列
    std::vector<MatchResult> SolveAll();

private:
    std::vector<MatchResult> SolveDungeonQueue(uint32_t dungeonId,
                                                  const std::vector<MatchQueue::MatchEntry>& queue);
    std::vector<MatchResult> SolveBattlegroundQueue(uint32_t bgId,
                                                      const std::vector<MatchQueue::MatchEntry>& queue);
    std::vector<MatchResult> SolveArenaQueue(uint32_t arenaId,
                                                const std::vector<MatchQueue::MatchEntry>& queue);
};

// 副本匹配算法：角色分配 + 等级范围 + 队组约束
std::vector<MatchResult> MatchSolver::SolveDungeonQueue(
    uint32_t dungeonId, const std::vector<MatchEntry>& queue)
{
    std::vector<MatchResult> results;

    // 1. 分类：Tank / Healer / DPS（按排队时间排序——早排优先）
    std::vector<const MatchEntry*> tanks, healers, dpsList;
    for (auto& e : queue) {
        if (e.accepted) continue;  // 跳过已确认但未处理的
        if (e.role == 0) tanks.push_back(&e);
        else if (e.role == 1) healers.push_back(&e);
        else dpsList.push_back(&e);
    }

    // 2. Greedy 匹配：每次取一个 Tank + Healer + 3 DPS
    size_t ti = 0, hi = 0, di = 0;
    while (ti < tanks.size() && hi < healers.size() && di + 2 < dpsList.size()) {
        const auto* tank   = tanks[ti];
        const auto* healer = healers[hi];

        // 检查等级范围：所有成员在 [baseLevel-3, baseLevel+3] 内
        int baseLevel = tank->level;
        auto levelOk = [baseLevel](const MatchEntry* e) {
            return std::abs(e->level - baseLevel) <= 3;
        };

        if (!levelOk(healer)) { hi++; continue; }

        // 收集 3 个 DPS
        std::vector<const MatchEntry*> dpsGroup;
        size_t dCursor = di;
        while (dpsGroup.size() < 3 && dCursor < dpsList.size()) {
            if (levelOk(dpsList[dCursor]) && !dpsList[dCursor]->accepted)
                dpsGroup.push_back(dpsList[dCursor]);
            dCursor++;
        }

        if (dpsGroup.size() < 3) break;  // 不够 DPS——本批无法匹配

        // 3. 匹配成功
        MatchResult result;
        result.type = MatchQueue::MatchType::Dungeon;
        result.dungeonId = dungeonId;
        result.players = {tank->player, healer->player,
                          dpsGroup[0]->player, dpsGroup[1]->player, dpsGroup[2]->player};
        result.averageMmr = (tank->mmr + healer->mmr +
                             dpsGroup[0]->mmr + dpsGroup[1]->mmr + dpsGroup[2]->mmr) / 5;
        results.push_back(result);

        ti++; hi++;
        di = dCursor;  // 剩余的 DPS 留给下一组
    }

    return results;
}

4.2 战场匹配（20v20 / 40v40）

std::vector<MatchResult> MatchSolver::SolveBattlegroundQueue(
    uint32_t bgId, const std::vector<MatchEntry>& queue)
{
    // 按 MMR 排序
    auto sorted = queue;
    std::sort(sorted.begin(), sorted.end(),
              [](auto& a, auto& b) { return a.mmr < b.mmr; });

    const int teamSize = 20;  // 或从配置读取

    if (sorted.size() < teamSize * 2) return {};

    // 蛇形分配 (Snake Draft)：MMR 排序后交替分配，保证双方实力均衡
    // Team A: index 0,3,4,7,8,11,12,15,16,19,20,23...
    // Team B: index 1,2,5,6,9,10,13,14,17,18,21,22...
    std::vector<Entity> teamA, teamB;
    bool toA = true;
    int count = 0;

    for (size_t i = 0; i < teamSize * 2; i++) {
        if (toA) teamA.push_back(sorted[i].player);
        else    teamB.push_back(sorted[i].player);
        count++;
        if (count == 2) { toA = !toA; count = 0; }
    }

    MatchResult result;
    result.type = MatchQueue::MatchType::Battleground;
    result.battlegroundId = bgId;
    result.players = teamA;
    result.players.insert(result.players.end(), teamB.begin(), teamB.end());
    return {result};
}

4.3 竞技场匹配（严格 MMR）

std::vector<MatchResult> MatchSolver::SolveArenaQueue(
    uint32_t arenaId, const std::vector<MatchEntry>& queue)
{
    // 竞技场使用 MMR Bucket：将玩家按 100 分一段分桶
    // 优先从同桶匹配，桶空则扩大到相邻桶
    constexpr int kBucketSize = 100;
    constexpr int kMaxBucketExpand = 3;  // 最多扩展到 ±3 个桶（±300 MMR）

    std::unordered_map<int, std::vector<const MatchEntry*>> buckets;
    for (auto& e : queue) {
        if (e.accepted) continue;
        buckets[e.mmr / kBucketSize].push_back(&e);
    }

    std::vector<MatchResult> results;

    // 按 partyId 分组（同一队伍的玩家一起匹配）
    // 简化：遍历所有桶，尝试找 MMR 接近的对手

    for (int bucketKey = 0; bucketKey < 30; bucketKey++) {  // MMR 0-3000
        auto it = buckets.find(bucketKey);
        if (it == buckets.end() || it->second.size() < 1) continue;

        // 找对手桶（从当前桶开始，逐步扩大）
        for (int expand = 0; expand <= kMaxBucketExpand; expand++) {
            for (int opponentBucket : {bucketKey + expand, bucketKey - expand}) {
                if (opponentBucket == bucketKey && expand == 0) continue;  // 同桶不限 MMR
                auto opIt = buckets.find(opponentBucket);
                if (opIt == buckets.end() || opIt->second.size() < 1) continue;

                if (it->second.size() >= 1 && opIt->second.size() >= 1) {
                    MatchResult result;
                    result.type = MatchQueue::MatchType::Arena;
                    result.arenaId = arenaId;
                    result.players = {it->second.back()->player, opIt->second.back()->player};
                    result.averageMmr = (it->second.back()->mmr + opIt->second.back()->mmr) / 2;
                    results.push_back(result);
                    it->second.pop_back();
                    opIt->second.pop_back();
                    break;
                }
            }
        }
    }

    return results;
}

6. DasLang 侧入口

// match_handler.das — 匹配系统入口
def HandleMatchQueue(player : Entity; matchType : int; specificId : uint32)
{
    // 校验状态
    if (IsInCombat(player)) { SendMatchRsp(player, MATCH_ERR_IN_COMBAT); return }
    if (IsInMatch(player))   { SendMatchRsp(player, MATCH_ERR_ALREADY_IN_MATCH); return }
    if (HasMatchPenalty(player)) {
        let remaining = GetMatchPenaltyRemaining(player)
        SendMatchRsp(player, MATCH_ERR_PENALTY, remaining); return
    }

    // 组装匹配条目
    let entry = MatchEntry{
        player     = player,
        playerId   = GetPlayerId(player),
        matchType  = matchType,
        dungeonId  = matchType == MATCH_DUNGEON ? specificId : 0,
        battlegroundId = matchType == MATCH_BG ? specificId : 0,
        arenaId    = matchType == MATCH_ARENA ? specificId : 0,
        level      = GetPlayerLevel(player),
        mmr        = GetPlayerMMR(player),
        role       = GetPlayerRole(player),
        partyId    = GetPlayerPartyId(player),
        enqueueTime = GetTickTime()
    }

    if (!MatchQueue.Enqueue(entry)) {
        SendMatchRsp(player, MATCH_ERR_ALREADY_IN_QUEUE); return
    }

    SendMatchQueueRsp(player, MATCH_OK)
}

7. 匹配协议

message MatchQueueReq {
    int32  match_type = 1;         // 0=Dungeon 1=Battleground 2=Arena
    uint32 specific_id = 2;        // dungeonId / bgId / arenaId
}
message MatchQueueRsp {
    MatchErrorCode error = 1;
    int32  penalty_seconds = 2;    // 处罚剩余时间（如有）
}

message MatchStatusNtf {
    int32  position = 1;           // 队列位置（1-based）
    int32  estimated_seconds = 2;  // 预估等待时间
}

message MatchFoundNtf {
    uint64 match_id = 1;
    repeated MatchMember members = 2;
}
message MatchMember {
    uint32 player_id = 1;
    string player_name = 2;
    int32  level = 3;
    int32  role = 4;
}

message MatchAcceptReq { uint64 match_id = 1; }
message MatchAcceptRsp {
    MatchErrorCode error = 1;
    uint32 scene_id = 2;           // 成功时——目标场景 ID
}

message MatchCancelReq {}
message MatchTimeoutNtf {}

enum MatchErrorCode {
    MATCH_OK                 = 0;
    MATCH_ERR_IN_COMBAT      = 1;
    MATCH_ERR_ALREADY_IN_MATCH = 2;
    MATCH_ERR_ALREADY_IN_QUEUE = 3;
    MATCH_ERR_PENALTY        = 4;
    MATCH_ERR_NOT_IN_QUEUE   = 5;
    MATCH_ERR_INVALID_DUNGEON = 6;
}
