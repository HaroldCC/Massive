15. 数据持久化 · 代码生成 + DBWorkerPool + PgBouncer

15.1 总体架构

方案 2：Schema → C++ 代码生成。C++ 层通过 Python 脚本从 PgSQL information_schema 自动生成 类型安全的 DAO 代码——开发者不写 SQL 字符串。底层 DBWorkerPool 持参数化 SQL + libpq → PgBouncer → PostgreSQL。 不设独立 DBServer 进程。

数据访问的两条路径（殊途同归——最终都走 DBWorkerPool）：

C++ 引擎层:
  db::Range<PlayersTable>()           ← 类型安全查询（由 Python 生成的 header）
      .Where(PlayersTable::PlayerId == 1001)
      .Select(PlayersTable::Name, PlayersTable::Level)
      .SingleOrDefault()
     → BuildSQL() → DBWorkerPool.AsyncQuery(sql, params, callback)

DasLang 脚本层:
  table("players")
      |> where_($(r) => r.player_id == 1001)
      |> select($(r) => {r.name, r.level})
      |> single_or_default()
     → AstCallMacro 展开 → BuildSQL() → DBWorkerPool.AsyncQuery(sql, params, callback)

复杂查询（JOIN / 子查询）:
  RuntimeQuery("SELECT ... JOIN ... WHERE ...", {params})  ← escape hatch

15.2.2 Column 实现——operator== 返回 Expression，不是 bool

// TableSchema.h —— 核心模板
template<typename T>
struct Column {
    const char* name;
    uint32_t flags;

    // ⚡ 关键: operator== 返回 Expression 而非 bool——这是 C++ 能实现 LINQ 语法的根基
    constexpr auto operator==(T value) const {
        return Expression<T>{name, Op::EQ, value};
    }

    constexpr auto operator<(T value)  const { return Expression<T>{name, Op::LT, value}; }
    constexpr auto operator>(T value)  const { return Expression<T>{name, Op::GT, value}; }
    constexpr auto operator<=(T value) const { return Expression<T>{name, Op::LE, value}; }
    constexpr auto operator>=(T value) const { return Expression<T>{name, Op::GE, value}; }
};

// Expression 支持 && 组合
template<typename T>
struct Expression {
    const char* columnName;
    Op op;
    T value;

    // WHERE a = 1 AND b > 5 → Expression::operator&&
    auto operator&&(const auto& other) const {
        return CompoundExpr{std::vector{this->clone(), other.clone()}, LogicalOp::AND};
    }
};

15.3 db::Range<T> —— 类型安全查询

// DBRange.h — 类型安全查询构建器
template<typename TableSchema>
class DBRange {
public:
    DBRange() : _tableName(TableSchema::kTableName) {}

    // ⚡ Where 接受 Expression——由 Column::operator== 返回
    template<typename... Exprs>
    auto& Where(auto expr, Exprs... more) {
        AddExpression(expr);
        if constexpr (sizeof...(more) > 0) return Where(more...);
        return *this;
    }

    // ⚡ Select 接受 Column 对象——编译期确定字段列表
    template<typename... Cols>
    auto& Select(Cols... cols) {
        (_selects.push_back({cols.name, TypeName<typename Cols::ValueType>()}), ...);
        return *this;
    }

    auto& OrderBy(auto col)     { _orderBy = col.name; _orderDesc = false; return *this; }
    auto& OrderByDesc(auto col) { _orderBy = col.name; _orderDesc = true;  return *this; }
    auto& Limit(int n)          { _limit = n; return *this; }

    // ⚡ 执行操作——BuildSQL() + 提交 DBWorkerPool + 结果反序列化到强类型
    template<typename Callback>
    void ToArray(Callback&& callback);      // 异步回调：vector<Row>

    template<typename Callback>
    void SingleOrDefault(Callback&& callback);  // Row or null

    // 写入操作（同步提交——立即投递给 Worker，不等结果）
    void Insert(auto&... rows);
    void Update(auto&& setExpr);   // 必须前有 Where()
    void Delete();                   // 必须前有 Where()

private:
    std::string _tableName;
    std::vector<Condition> _conditions;
    std::vector<SelectColumn> _selects;
    std::string _orderBy;
    bool _orderDesc = false;
    int _limit = -1;
    DBWorkerPool* _pool = DBWorkerPool::Instance();
};

15.4 C++ 层使用——完整示例

// ========== 查询 ==========
// 单行查询——类型安全，字段名是编译期常量
db::Range<PlayersTable>()
    .Where(PlayersTable::PlayerId == 1001)
    .SingleOrDefault([](std::optional<PlayerRow> player) {
        if (player) LoadPlayerIntoWorld(*player);
    });

// 多条件查询
auto healers = db::Range<PlayersTable>()
    .Where(PlayersTable::ClassId == CLASS_PRIEST, PlayersTable::Level >= 60)
    .OrderBy(PlayersTable::Level)
    .Limit(100);

// ========== 写入 ==========
// INSERT——结构化初始化，字段遗漏 = 编译报错
db::Range<PlayersTable>().Insert(
    PlayerRow{
        .AccountId = 5001,
        .Name      = "DarkMage",
        .ClassId   = CLASS_MAGE,
        .Level     = 1,
        .SceneId   = 1001,
        .PosX      = 100.0f,
        .PosY      = 50.0f,
        .PosZ      = 0.0f
    }
);

// UPDATE
db::Range<PlayersTable>()
    .Where(PlayersTable::PlayerId == 1001)
    .Update(UpdateSet{}.Set(PlayersTable::Level, 61)
                       .Set(PlayersTable::Gold,   99999LL));

// DELETE
db::Range<GuildsTable>()
    .Where(GuildsTable::GuildId == 42)
    .Delete();
🧠 设计要点：为什么 Column::operator== 返回 Expression 而非 bool
这是 C++ 实现类型安全查询的核心技巧。如果 operator== 返回 bool， Where(PlayersTable::Level >= 60) 会在编译期被求值为 Where(true)——丢失了列名和条件。 返回 Expression 对象则保留了所有编译期信息——列名 "level"、运算符 >=、值 60—— Where() 收集这些信息，ToArray() 时拼接为参数化 SQL。效果和 C# LINQ-to-SQL 一致。

15.5 DBWorkerPool —— 异步执行层

// DBWorkerPool.h — 所有需要 DB 的进程共用（common 库）
class DBWorkerPool {
public:
    static DBWorkerPool* Instance() { static DBWorkerPool inst; return &inst; }

    DBWorkerPool(int workerCount, const std::string& connString);

    // ⚡ 参数化 SQL——防注入。SQL 由 db::Range 或 DBQuery 的 BuildSQL() 生成
    void AsyncQuery(std::string sql,
                  std::vector<DbValue> params,
                  std::function<void(const DBResult&)> callback);

    void ProcessCallbacks();  // LogicThread 每 Tick 调用

private:
    struct Worker {
        PGconn* conn;
        std::thread thread;
        MpscQueue<DBTask> taskQueue;
    };
    std::vector<Worker> _workers;
    MpscQueue<DBCallback> _completedCallbacks;
};

// Worker 线程——使用参数化查询（PQexecParams），非无参 PQexec
void DBWorkerPool::Worker::Run() {
    while (!_stop) {
        for (auto& task : taskQueue.DrainAll()) {
            auto result = PQexecParams(conn, task.sql.c_str(),
                int(task.params.size()), nullptr,
                task.paramValues(), task.paramLengths(), task.paramFormats(),
                0);  // 0 = text format

            if (PQresultStatus(result) == PGRES_TUPLES_OK ||
                PQresultStatus(result) == PGRES_COMMAND_OK) {
                _pool->_completedCallbacks.Enqueue({task.callback, DBResult{result}});
            } else {
                LOG_ERROR(0, "DB error: {}", PQresultErrorMessage(result));
            }
            PQclear(result);
        }
    }
}

15.7 RuntimeQuery —— 复杂查询 escape hatch

对于 LINQ 管道链无法表达的复杂查询（跨表 JOIN、子查询、窗口函数、CTE），保留参数化 SQL 接口。 这是 DBWorkerPool 的唯一公开 SQL 入口——正常的 CRUD 都应走 db::Range<T> 或 DasLang LINQ。 

// C++ 侧
auto result = RuntimeQuery(
    "SELECT p.player_id, p.name, SUM(c.damage) as total_dps
     FROM players p
     JOIN combat_log c ON p.player_id = c.player_id
     WHERE p.guild_id = $1 AND c.log_time > $2
     GROUP BY p.player_id, p.name
     ORDER BY total_dps DESC LIMIT 10",
    {guildId, GetTickTime() - 3600}
);

// DasLang 侧
def GetTopDpsByGuild(guildId : uint32) : array<PlayerDpsInfo>
{
    var result : array<PlayerDpsInfo>
    RuntimeQuery(
        "SELECT ... JOIN ... WHERE ... GROUP BY ...",
        [guildId, GetTickTime() - 3600]
    ) $(var rows : array<DBRow>) { ... }
    return result
}
