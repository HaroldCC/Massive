6.2 WorldServer 侧：WorldSession::Send

// WorldSession — WorldServer 出站
void WorldSession::Send(std::vector<char> body) {
    // 1. 加密 Body（CryptoSession 内维护 _sendSequence）
    auto encrypted = crypto.Encrypt(std::move(body));

    // 2. 包装 InternalHeader + PacketHeader + 密文
    InternalHeader ih;
    ih.sessionID = sessionID;
    // 3. 写入 gateServerID 对应的 Gate 内部 TCP 连接
    GateConnectionMgr::SendToGate(gateServerID, ih, std::move(encrypted));
}

6.3 GateServer 侧：收 World 包 → 转发 Client

// GateServer.cpp — WorldServer → Client 转发
void GateServer::OnWorldPacket(GateConnectionPtr worldConn,
                                 const std::vector<char>& data) {
    // 1. 解析 InternalHeader → sessionId
    auto header = ParseInternalHeader(data);

    // 2. 查 session → clientSocket
    auto it = _sessionRoutes.find(header.sessionId);
    if (it == _sessionRoutes.end()) return;  // 客户端已断开

    auto& route = it->second;

    // 3. 包装为 Wire 格式: [Length | MsgID | SessionId | EncryptedBody]
    //    Body 部分原样透传——Gate 不解密、不解析
    std::vector<char> wire;
    AppendWireHeader(wire, data.payload());

    // 4. 发送到客户端
    route.clientSession->SendToClient(std::move(wire));
}

6.4 Gate 侧写回——同现有设计

GateSession 的 SendToClient 与 §2 的 Session::Send 一致——加锁写 _writeBuffer + async_write 链式写回。 

void GateSession::SendToClient(std::vector<char> data) {
    bool needWrite = false;
    { std::lock_guard lock(_writeMutex);
      _writeBuffer.insert(_writeBuffer.end(), data.begin(), data.end());
      if (!_writing) { needWrite = true; _writing = true; } }
    if (needWrite) DoWrite();
}

6.5 连接到多个 Gate——WorldServer 侧 GateConnection 管理

// WorldServer.h — 管理到各 GateServer 的连接
class GateConnectionMgr {
public:
    struct GateConnection {
        uint16_t gateId;
        std::shared_ptr<TCPSocket> socket;  // 到 Gate 的内部 TCP
    };

    // Gate 建立世界时注册
    void AcceptGateConnection(uint16_t gateId, std::shared_ptr<TCPSocket> socket);

    // 业务消息：加密后带 InternalHeader 发送
    void SendToGate(uint16_t gateId, const InternalHeader& hdr, ByteBuffer payload);

    // 控制消息：走 RPC Notify 协议统一通道（DisconnectNtf/SessionRebindReq 等）
    void SendControlToGate(uint16_t gateId, uint32_t msgID, const google::protobuf::Message& msg);

private:
    std::unordered_map<uint16_t, std::unique_ptr<GateConnection>> _gateConns;
};
