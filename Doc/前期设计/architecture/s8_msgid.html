8.1 核心设计：MsgID 数组索引 + 模板注册

std::array<Handler, 4096> 做 O(1) 查表，msgId 直接作为数组下标。每新增一个消息一行 Register。

class MessageDispatcher {
public:
    struct Handler {
        std::function<void(WorldEntity, const std::vector<char>&)> fn;
    };

    template<typename T>
    void Register(uint32_t msgId, std::function<void(Entity, const T&)> handler) {
        // 重复注册检测
        if (_handlers[msgId].fn) {
            LOG_WARN(0, "MsgID {} already registered, overwriting", msgId);
        }
        _handlers[msgId].fn = [handler](Entity we, const std::vector<char>& body) {
            T msg;
            if (!msg.ParseFromArray(body.data(), int(body.size()))) {
                LOG_ERROR(0, "Protobuf parse failed for MsgID {} ({} bytes)", msgId, body.size());
                return;  // 解析失败，跳过并记录日志
            }
            handler(we, msg);
        };
    }

    void Dispatch(WorldEntity we, uint32_t msgId, const std::vector<char>& body) {
        if (msgId < _handlers.size() && _handlers[msgId].fn)
            _handlers[msgId].fn(we, body);
    }
private:
    std::array<Handler, 4096> _handlers;
};

// 注册示例
dispatcher.Register<MoveReq>(MSG_MOVE_REQ, [](WorldEntity we, const MoveReq& msg) {
    HandleMove(we, msg.x, msg.y);
});
