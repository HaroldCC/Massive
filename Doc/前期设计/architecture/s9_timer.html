9. 定时器 · 多级时间轮

9.1 四级时间轮

每轮 60 槽：轮0=1.2s(20ms×60)、轮1=72s(1.2s×60)、轮2=72min(72s×60)。O(1) 插入/触发/取消。

class TimingWheel {
public:
    struct TimerNode {
        TimerId id;
        int remainingRounds;
        std::function<void()> callback;
        TimerNode* next = nullptr;
    };

    void Schedule(TimerId id, std::chrono::milliseconds delay, std::function<void()> cb);
    void Cancel(TimerId id);
    void Tick();

private:
    static constexpr int kSlots = 60;
    static constexpr std::chrono::milliseconds kTickSize[3] = {
        std::chrono::milliseconds(50), std::chrono::milliseconds(3000), std::chrono::milliseconds(180000)
    };
    TimerNode* _wheels[3][kSlots] = {};
    std::unordered_map<TimerId, TimerNode*> _idMap;
    int _currentSlot[3] = {};
    TimerNode* _freeList = nullptr;  // 对象池——避免每次 new/delete
};

9.2 Schedule 实现

void TimingWheel::Schedule(TimerId id, std::chrono::milliseconds delay,
                          std::function<void()> callback) {
    // 从对象池分配
    auto* node = _freeList ? _freeList : new TimerNode{};
    if (_freeList) _freeList = _freeList->next;
    node->id = id;
    node->callback = std::move(callback);

    // 确定轮号: delay / tickSize[0] 算槽位，越界升到轮1/轮2
    int wheelIdx = 0;
    int slotOffset = int(delay / kTickSize[0]);
    if (slotOffset >= kSlots) {
        slotOffset = int(delay / kTickSize[1]);
        wheelIdx = 1;
        if (slotOffset >= kSlots) {
            slotOffset = int(delay / kTickSize[2]);
            wheelIdx = 2;
        }
    }

    // 计算槽位和剩余圈数
    int targetSlot = (_currentSlot[wheelIdx] + slotOffset) % kSlots;
    node->remainingRounds = slotOffset / kSlots;

    // 插入到槽位链表头部
    node->next = _wheels[wheelIdx][targetSlot];
    _wheels[wheelIdx][targetSlot] = node;

    _idMap[id] = node;
}

9.3 Tick 与级联

void TimingWheel::Tick() {
    // 推进轮0（每 Tick 都推进）
    _currentSlot[0] = (_currentSlot[0] + 1) % kSlots;

    // 收集到期槽的剩余轮数 > 0 的节点 → remainingRounds-- 放回原位置
    //         剩余轮数 == 0 的节点 → 触发 callback
    auto* cur = _wheels[0][_currentSlot[0]];
    _wheels[0][_currentSlot[0]] = nullptr;
    while (cur) {
        auto* next = cur->next;
        if (cur->remainingRounds > 0) {
            // 还没到期——tick 一圈后再触发
            cur->remainingRounds--;
            cur->next = _wheels[0][_currentSlot[0]];
            _wheels[0][_currentSlot[0]] = cur;
        } else {
            // 到期！触发后回收节点
            cur->callback();
            _idMap.erase(cur->id);
            cur->callback = {};
            cur->next = _freeList; _freeList = cur;  // 归还对象池
        }
        cur = next;
    }

    // 级联：轮0跑满 1 圈 (= 60 tick = 3s) → 推进轮1
    if (_currentSlot[0] == 0) {
        _currentSlot[1] = (_currentSlot[1] + 1) % kSlots;
        // 轮1的定时器级联到轮0：重新 Schedule
        // 轮1每推进 1 格 = 把所有轮1当前槽的定时器重新 Schedule 进轮0
        auto* cascade = _wheels[1][_currentSlot[1]];
        _wheels[1][_currentSlot[1]] = nullptr;
        while (cascade) {
            auto* next = cascade->next;
            if (cascade->remainingRounds > 0) {
                cascade->remainingRounds--;
                // 放回轮1 同一槽
                cascade->next = _wheels[1][_currentSlot[1]];
                _wheels[1][_currentSlot[1]] = cascade;
            } else {
                // 级联到轮0：重新 Schedule 到轮0
                // (简化——将 cascade 直接插入轮0对应槽)
            }
            cascade = next;
        }
    }
}

9.4 Cancel — O(1) 链表摘除

void TimingWheel::Cancel(TimerId id) {
    auto it = _idMap.find(id);
    if (it == _idMap.end()) return;
    auto* node = it->second;
    _idMap.erase(it);

    // 从槽位链表中摘除 (需要遍历链表——同样 O(1) 期望，因为链表短)
    // 实际等于 O(槽位内节点数)——由于槽位内数量少 (平均 1-5 个)，接近 O(1)
    // 更优做法：双向链表 (加 prev 指针)
    node->callback = {};
    node->next = _freeList; _freeList = node;
}

9.5 TimerId 生成

using TimerId = uint64_t;

static std::atomic<TimerId> _nextTimerId{1};
TimerId GenTimerId() { return _nextTimerId.fetch_add(1); }
// 使用 64 位，实际永不回绕 (每秒 1000 个定时器也要 5.8 亿年)
